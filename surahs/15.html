Tajweed keyrules in this surah
  • Madd Laazimمد لازمHold 6 counts. Obligatory.
  • Madd Muttasilمد متصلHold 4 to 5 counts.
  • Madd Munfasilمد منفصلHold 4 to 5 counts.
  • Madd al-Aaridمد عارضHold 2, 4 or 6 counts when stopping.
  • IqlabإقلابNoon turns into a meem sound.
  • Idghaam w/ Ghunnahإدغام بغنةMerge into the next letter with nasalisation.
  • Idghaam w/o Ghunnahإدغام بغير غنةMerge into the next letter, no nasalisation.
  • Idghaam Shafawiإدغام شفويMeem merges into meem.
  • Ikhfa Shafawiإخفاء شفويHide the meem before a baa.
  • IkhfaإخفاءHide the noon with a light nasal sound.
  • GhunnahغنةNasal sound, 2 counts.
  • QalqalahقلقلةEcho the letter with a slight bounce.
  • Lam Shamsiyyahلام شمسيةSilent laam, merged into the next letter.
  • Hamzat al-Waslهمزة وصلSkipped when joined to the word before.
  • Madd Tabee'iمد طبيعيNatural madd. Hold 2 counts.
Al-Hijr الحجر Meccan · 99 ayat
الٓرۚalif laam raa تِلْكَtilka ءَايَٰتُaayaatu ٱلْكِتَٰبِl-kitaabi وَقُرْءَانٍwaqur'aanin مُّبِينٍmubeen١ رُّبَمَاrrubamaa يَوَدُّyawaddu ٱلَّذِينَl-ladheena كَفَرُواْkafarooaa لَوْlaw كَانُواْkaanooaa مُسْلِمِينَmuslimeen٢ ذَرْهُمْdharhum يَأْكُلُواْya'kulooaa وَيَتَمَتَّعُواْwayatamatta'ooaa وَيُلْهِهِمُwayulhihimu ٱلْأَمَلُۖl-'amalu فَسَوْفَfasawfa يَعْلَمُونَya'lamoon٣ وَمَآwamaa أَهْلَكْنَاahlaknaa مِنmin قَرْيَةٍqaryahin إِلَّاillaa وَلَهَاwalahaa كِتَابٌkitaabun مَّعْلُومٌma'loom٤ مَّاmmaa تَسْبِقُtasbiqu مِنْmin أُمَّةٍummahin أَجَلَهَاajalahaa وَمَاwamaa أَرْسَلْنَاarsalnaa مِنmin قَبْلِكَqablika فِيfee شِيَعِshiya'i ٱلْأَوَّلِينَl-'awwaleen١٠ وَمَاwamaa يَأْتِيهِمya'teehim مِّنmin رَّسُولٍrasoolin إِلَّاillaa كَانُواْkaanooaa بِهِۦbihee يَسْتَهْزِءُونَyastahzi'oon١١ كَذَٰلِكَkadhaalika نَسْلُكُهُۥnaslukuhoo فِيfee قُلُوبِquloobi ٱلْمُجْرِمِينَl-mujrimeen١٢ لَاlaa يُؤْمِنُونَyu'minoona بِهِۦbihee وَقَدْwaqad خَلَتْkhalat سُنَّةُsunnahu ٱلْأَوَّلِينَl-'awwaleen١٣ وَلَوْwalaw فَتَحْنَاfatahnaa عَلَيْهِم'alayhim بَابًاbaaban مِّنَmina ٱلسَّمَآءِs-samaa'i فَظَلُّواْfazallooaa فِيهِfeehi يَعْرُجُونَya'rujoon١٤ لَقَالُوٓاْlaqaalooaa إِنَّمَاinnamaa سُكِّرَتْsukkirat أَبْصَٰرُنَاabsaarunaa بَلْbal نَحْنُnahnu قَوْمٌqawmun مَّسْحُورُونَmashooroon١٥ وَلَقَدْwalaqad جَعَلْنَاja'alnaa فِيfee ٱلسَّمَآءِs-samaa'i بُرُوجًاburoojan وَزَيَّنَّٰهَاwazayyannaahaa لِلنَّٰظِرِينَlinnaazireen١٦ وَحَفِظْنَٰهَاwahafiznaahaa مِنmin كُلِّkulli شَيْطَٰنٍshaytaanin رَّجِيمٍrajeem١٧ إِلَّاillaa مَنِmani ٱسْتَرَقَs-taraqa ٱلسَّمْعَs-sam'a فَأَتْبَعَهُۥfa'atba'ahoo شِهَابٌshihaabun مُّبِينٌmubeen١٨ وَٱلْأَرْضَwal'arda مَدَدْنَٰهَاmadadnaahaa وَأَلْقَيْنَاwa'alqaynaa فِيهَاfeehaa رَوَٰسِيَrawaasiya وَأَنۢبَتْنَاwa'anbatnaa فِيهَاfeehaa مِنmin كُلِّkulli شَيْءٍshay'in مَّوْزُونٍmawzoon١٩ وَجَعَلْنَاwaja'alnaa لَكُمْlakum فِيهَاfeehaa مَعَٰيِشَma'aayisha وَمَنwaman لَّسْتُمْlastum لَهُۥlahoo بِرَٰزِقِينَbiraaziqeen٢٠ وَإِنwa'in مِّنmin شَيْءٍshay'in إِلَّاillaa عِندَنَا'indanaa خَزَآئِنُهُۥkhazaa'inuhoo وَمَاwamaa نُنَزِّلُهُۥٓnunazziluhoo إِلَّاillaa بِقَدَرٍbiqadarin مَّعْلُومٍma'loom٢١ وَأَرْسَلْنَاwa'arsalnaa ٱلرِّيَٰحَr-riyaaha لَوَٰقِحَlawaaqiha فَأَنزَلْنَاfa'anzalnaa مِنَmina ٱلسَّمَآءِs-samaa'i مَآءًmaa'an فَأَسْقَيْنَٰكُمُوهُfa'asqaynaakumoohu وَمَآwamaa أَنتُمْantum لَهُۥlahoo بِخَٰزِنِينَbikhaazineen٢٢ وَإِنَّاwa'innaa لَنَحْنُlanahnu نُحْيِۦnuhyee وَنُمِيتُwanumeetu وَنَحْنُwanahnu ٱلْوَٰرِثُونَl-waarithoon٢٣ وَلَقَدْwalaqad عَلِمْنَا'alimnaa ٱلْمُسْتَقْدِمِينَl-mustaqdimeena مِنكُمْminkum وَلَقَدْwalaqad عَلِمْنَا'alimnaa ٱلْمُسْتَئْخِرِينَl-musta'khireen٢٤ وَإِنَّwa'inna رَبَّكَrabbaka هُوَhuwa يَحْشُرُهُمْۚyahshuruhum إِنَّهُۥinnahoo حَكِيمٌhakeemun عَلِيمٌ'aleem٢٥ وَلَقَدْwalaqad خَلَقْنَاkhalaqnaa ٱلْإِنسَٰنَl-'insaana مِنmin صَلْصَٰلٍsalsaalin مِّنْmin حَمَإٍhama'in مَّسْنُونٍmasnoon٢٦ وَٱلْجَآنَّwaljaanna خَلَقْنَٰهُkhalaqnaahu مِنmin قَبْلُqablu مِنmin نَّارِnaari ٱلسَّمُومِs-samoom٢٧ وَإِذْwa'idh قَالَqaala رَبُّكَrabbuka لِلْمَلَٰٓئِكَةِlilmalaa'ikahi إِنِّيinnee خَٰلِقُۢkhaaliqu بَشَرًاbasharan مِّنmin صَلْصَٰلٍsalsaalin مِّنْmin حَمَإٍhama'in مَّسْنُونٍmasnoon٢٨ فَإِذَاfa'idhaa سَوَّيْتُهُۥsawwaytuhoo وَنَفَخْتُwanafakhtu فِيهِfeehi مِنmin رُّوحِيroohee فَقَعُواْfaqa'ooaa لَهُۥlahoo سَٰجِدِينَsaajideen٢٩ فَسَجَدَfasajada ٱلْمَلَٰٓئِكَةُl-malaa'ikahu كُلُّهُمْkulluhum أَجْمَعُونَajma'oon٣٠ إِلَّآillaa إِبْلِيسَibleesa أَبَىٰٓabaa أَنan يَكُونَyakoona مَعَma'a ٱلسَّٰجِدِينَs-saajideen٣١ قَالَqaala يَٰٓإِبْلِيسُyaa'ibleesu مَاmaa لَكَlaka أَلَّاallaa تَكُونَtakoona مَعَma'a ٱلسَّٰجِدِينَs-saajideen٣٢ قَالَqaala لَمْlam أَكُنakun لِّأَسْجُدَli'asjuda لِبَشَرٍlibasharin خَلَقْتَهُۥkhalaqtahoo مِنmin صَلْصَٰلٍsalsaalin مِّنْmin حَمَإٍhama'in مَّسْنُونٍmasnoon٣٣ قَالَqaala فَٱخْرُجْfakhruj مِنْهَاminhaa فَإِنَّكَfa'innaka رَجِيمٌrajeem٣٤ وَإِنَّwa'inna عَلَيْكَ'alayka ٱللَّعْنَةَl-la'naha إِلَىٰilaa يَوْمِyawmi ٱلدِّينِd-deen٣٥ قَالَqaala رَبِّrabbi فَأَنظِرْنِيٓfa'anzirnee إِلَىٰilaa يَوْمِyawmi يُبْعَثُونَyub'athoon٣٦ قَالَqaala لَأُزَيِّنَنَّla'uzayyinanna لَهُمْlahum فِيfee ٱلْأَرْضِl-'ardi وَلَأُغْوِيَنَّهُمْwala'ughwiyannahum أَجْمَعِينَajma'een٣٩ إِلَّاillaa عِبَادَكَ'ibaadaka مِنْهُمُminhumu ٱلْمُخْلَصِينَl-mukhlaseen٤٠ فَٰعِلِينَfaa'ileen٧١ لَعَمْرُكَla'amruka إِنَّهُمْinnahum لَفِيlafee سَكْرَتِهِمْsakratihim يَعْمَهُونَya'mahoon٧٢ فَأَخَذَتْهُمُfa'akhadhathumu ٱلصَّيْحَةُs-sayhahu مُشْرِقِينَmushriqeen٧٣ فَجَعَلْنَاfaja'alnaa عَٰلِيَهَا'aaliyahaa سَافِلَهَاsaafilahaa وَأَمْطَرْنَاwa'amtarnaa عَلَيْهِمْ'alayhim حِجَارَةًhijaarahan مِّنmin سِجِّيلٍsijjeel٧٤ إِنَّinna فِيfee ذَٰلِكَdhaalika لَأٓيَٰتٍla'yaatin لِّلْمُتَوَسِّمِينَlilmutawassimeen٧٥ وَإِنَّهَاwa'innahaa لَبِسَبِيلٍlabisabeelin مُّقِيمٍmuqeem٧٦ إِنَّinna فِيfee ذَٰلِكَdhaalika لَأٓيَةًla'yahan لِّلْمُؤْمِنِينَlilmu'mineen٧٧ وَإِنwa'in كَانَkaana أَصْحَٰبُashaabu ٱلْأَيْكَةِl-'aykahi لَظَٰلِمِينَlazaalimeen٧٨ فَٱنتَقَمْنَاfantaqamnaa مِنْهُمْminhum وَإِنَّهُمَاwa'innahumaa لَبِإِمَامٍlabi'imaamin مُّبِينٍmubeen٧٩ وَلَقَدْwalaqad كَذَّبَkadhdhaba أَصْحَٰبُashaabu ٱلْحِجْرِl-hijri ٱلْمُرْسَلِينَl-mursaleen٨٠ وَءَاتَيْنَٰهُمْwa'aataynaahum ءَايَٰتِنَاaayaatinaa فَكَانُواْfakaanooaa عَنْهَا'anhaa مُعْرِضِينَmu'rideen٨١ وَكَانُواْwakaanooaa يَنْحِتُونَyanhitoona مِنَmina ٱلْجِبَالِl-jibaali بُيُوتًاbuyootan ءَامِنِينَaamineen٨٢ فَأَخَذَتْهُمُfa'akhadhathumu ٱلصَّيْحَةُs-sayhahu مُصْبِحِينَmusbiheen٨٣ فَمَآfamaa أَغْنَىٰaghnaa عَنْهُم'anhum مَّاmaa كَانُواْkaanooaa يَكْسِبُونَyaksiboon٨٤ وَمَاwamaa خَلَقْنَاkhalaqnaa ٱلسَّمَٰوَٰتِs-samaawaati وَٱلْأَرْضَwal'arda وَمَاwamaa بَيْنَهُمَآbaynahumaa إِلَّاillaa بِٱلْحَقِّۗbilhaqqi وَإِنَّwa'inna ٱلسَّاعَةَs-saa'aha لَأٓتِيَةٌۖla'tiyahun فَٱصْفَحِfasfahi ٱلصَّفْحَs-safha ٱلْجَمِيلَl-jameel٨٥ إِنَّinna رَبَّكَrabbaka هُوَhuwa ٱلْخَلَّٰقُl-khallaaqu ٱلْعَلِيمُl-'aleem٨٦ وَلَقَدْwalaqad ءَاتَيْنَٰكَaataynaaka سَبْعًاsab'an مِّنَmina ٱلْمَثَانِيl-mathaanee وَٱلْقُرْءَانَwalqur'aana ٱلْعَظِيمَl-'azeem٨٧ لَاlaa تَمُدَّنَّtamuddanna عَيْنَيْكَ'aynayka إِلَىٰilaa مَاmaa مَتَّعْنَاmatta'naa بِهِۦٓbihee أَزْوَٰجًاazwaajan مِّنْهُمْminhum وَلَاwalaa تَحْزَنْtahzan عَلَيْهِمْ'alayhim وَٱخْفِضْwakhfid جَنَاحَكَjanaahaka لِلْمُؤْمِنِينَlilmu'mineen٨٨ وَقُلْwaqul إِنِّيٓinnee أَنَاanaa ٱلنَّذِيرُn-nadheeru ٱلْمُبِينُl-mubeen٨٩ كَمَآkamaa أَنزَلْنَاanzalnaa عَلَى'alaa ٱلْمُقْتَسِمِينَl-muqtasimeen٩٠ ٱلَّذِينَalladheena جَعَلُواْja'alooaa ٱلْقُرْءَانَl-qur'aana عِضِينَ'ideen٩١ فَوَرَبِّكَfawarabbika لَنَسْئَلَنَّهُمْlanas'alannahum أَجْمَعِينَajma'een٩٢ عَمَّا'ammaa كَانُواْkaanooaa يَعْمَلُونَya'maloon٩٣ فَٱصْدَعْfasda' بِمَاbimaa تُؤْمَرُtu'maru وَأَعْرِضْwa'a'rid عَنِ'ani ٱلْمُشْرِكِينَl-mushrikeen٩٤ إِنَّاinnaa كَفَيْنَٰكَkafaynaaka ٱلْمُسْتَهْزِءِينَl-mustahzi'een٩٥ ٱلَّذِينَalladheena يَجْعَلُونَyaj'aloona مَعَma'a ٱللَّهِl-laahi إِلَٰهًاilaahan ءَاخَرَۚaakhara فَسَوْفَfasawfa يَعْلَمُونَya'lamoon٩٦ وَلَقَدْwalaqad نَعْلَمُna'lamu أَنَّكَannaka يَضِيقُyadeequ صَدْرُكَsadruka بِمَاbimaa يَقُولُونَyaqooloon٩٧ فَسَبِّحْfasabbih بِحَمْدِbihamdi رَبِّكَrabbika وَكُنwakun مِّنَmina ٱلسَّٰجِدِينَs-saajideen٩٨ وَٱعْبُدْwa'bud رَبَّكَrabbaka حَتَّىٰhattaa يَأْتِيَكَya'tiyaka ٱلْيَقِينُl-yaqeen٩٩
