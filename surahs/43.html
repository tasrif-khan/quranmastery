Tajweed keyrules in this surah
  • Madd Laazimمد لازمHold 6 counts. Obligatory.
  • Madd Muttasilمد متصلHold 4 to 5 counts.
  • Madd Munfasilمد منفصلHold 4 to 5 counts.
  • Madd al-Aaridمد عارضHold 2, 4 or 6 counts when stopping.
  • IqlabإقلابNoon turns into a meem sound.
  • Idghaam w/ Ghunnahإدغام بغنةMerge into the next letter with nasalisation.
  • Idghaam w/o Ghunnahإدغام بغير غنةMerge into the next letter, no nasalisation.
  • Idghaam Shafawiإدغام شفويMeem merges into meem.
  • Idghaam Mutajanisaynإدغام متجانسينLetters of the same articulation merge.
  • Ikhfa Shafawiإخفاء شفويHide the meem before a baa.
  • IkhfaإخفاءHide the noon with a light nasal sound.
  • GhunnahغنةNasal sound, 2 counts.
  • QalqalahقلقلةEcho the letter with a slight bounce.
  • Lam Shamsiyyahلام شمسيةSilent laam, merged into the next letter.
  • Hamzat al-Waslهمزة وصلSkipped when joined to the word before.
  • Silent letterحرف لا ينطقWritten but not pronounced.
  • Madd Tabee'iمد طبيعيNatural madd. Hold 2 counts.

أُلْقِيَulqiya عَلَيْهِ'alayhi أَسْوِرَةٌaswirahun مِّنmin ذَهَبٍdhahabin أَوْaw جَآءَjaa'a مَعَهُma'ahu ٱلْمَلَٰٓئِكَةُl-malaa'ikahu مُقْتَرِنِينَmuqtarineen٥٣ فَٱسْتَخَفَّfastakhaffa قَوْمَهُۥqawmahoo فَأَطَاعُوهُۚfa'ataa'oohu إِنَّهُمْinnahum كَانُواْkaanooaa قَوْمًاqawman فَٰسِقِينَfaasiqeen٥٤ فَلَمَّآfalammaa ءَاسَفُونَاaasafoonaa ٱنتَقَمْنَاn-taqamnaa مِنْهُمْminhum فَأَغْرَقْنَٰهُمْfa'aghraqnaahum أَجْمَعِينَajma'een٥٥ فَجَعَلْنَٰهُمْfaja'alnaahum سَلَفًاsalafan وَمَثَلًاwamathalan لِّلْأٓخِرِينَlil'khireen٥٦ ۞وَلَمَّاwalammaa ضُرِبَduriba ٱبْنُb-nu مَرْيَمَmaryama مَثَلًاmathalan إِذَاidhaa قَوْمُكَqawmuka مِنْهُminhu يَصِدُّونَyasiddoon٥٧ وَقَالُوٓاْwaqaalooaa ءَأَٰلِهَتُنَاa'aalihatunaa خَيْرٌkhayrun أَمْam هُوَۚhuwa مَاmaa ضَرَبُوهُdaraboohu لَكَlaka إِلَّاillaa جَدَلَۢاۚjadalaa بَلْbal هُمْhum قَوْمٌqawmun خَصِمُونَkhasimoon٥٨ إِنْin هُوَhuwa إِلَّاillaa عَبْدٌ'abdun أَنْعَمْنَاan'amnaa عَلَيْهِ'alayhi وَجَعَلْنَٰهُwaja'alnaahu مَثَلًاmathalan لِّبَنِيٓlibanee إِسْرَٰٓءِيلَisraa'eel٥٩ وَلَوْwalaw نَشَآءُnashaa'u لَجَعَلْنَاlaja'alnaa مِنكُمminkum مَّلَٰٓئِكَةًmalaa'ikahan فِيfee ٱلْأَرْضِl-'ardi يَخْلُفُونَyakhlufoon٦٠ وَإِنَّهُۥwa'innahoo لَعِلْمٌla'ilmun لِّلسَّاعَةِlissaa'ahi فَلَاfalaa تَمْتَرُنَّtamtarunna بِهَاbihaa وَٱتَّبِعُونِۚwattabi'ooni هَٰذَاhaadhaa صِرَٰطٌsiraatun مُّسْتَقِيمٌmustaqeem٦١ وَلَاwalaa يَصُدَّنَّكُمُyasuddannakumu ٱلشَّيْطَٰنُۖsh-shaytaanu إِنَّهُۥinnahoo لَكُمْlakum عَدُوٌّ'aduwwun مُّبِينٌmubeen٦٢ وَلَمَّاwalammaa جَآءَjaa'a عِيسَىٰ'eesaa بِٱلْبَيِّنَٰتِbilbayyinaati قَالَqaala قَدْqad جِئْتُكُمji'tukum بِٱلْحِكْمَةِbilhikmahi وَلِأُبَيِّنَwali'ubayyina لَكُمlakum بَعْضَba'da ٱلَّذِيl-ladhee تَخْتَلِفُونَtakhtalifoona فِيهِۖfeehi فَٱتَّقُواْfattaqooaa ٱللَّهَl-laaha وَأَطِيعُونِwa'atee'oon٦٣ إِنَّinna ٱللَّهَl-laaha هُوَhuwa رَبِّيrabbee وَرَبُّكُمْwarabbukum فَٱعْبُدُوهُۚfa'budoohu هَٰذَاhaadhaa صِرَٰطٌsiraatun مُّسْتَقِيمٌmustaqeem٦٤ فَٱخْتَلَفَfakhtalafa ٱلْأَحْزَابُl-'ahzaabu مِنۢmin بَيْنِهِمْۖbaynihim فَوَيْلٌfawaylun لِّلَّذِينَlilladheena ظَلَمُواْzalamooaa مِنْmin عَذَابِ'adhaabi يَوْمٍyawmin أَلِيمٍaleem٦٥ هَلْhal يَنظُرُونَyanzuroona إِلَّاillaa ٱلسَّاعَةَs-saa'aha أَنan تَأْتِيَهُمta'tiyahum بَغْتَةًbaghtahan وَهُمْwahum لَاlaa يَشْعُرُونَyash'uroon٦٦ ٱلْأَخِلَّآءُal'akhillaa'u يَوْمَئِذِۭyawma'idhi بَعْضُهُمْba'duhum لِبَعْضٍliba'din عَدُوٌّ'aduwwun إِلَّاillaa ٱلْمُتَّقِينَl-muttaqeen٦٧ يَٰعِبَادِyaa'ibaadi لَاlaa خَوْفٌkhawfun عَلَيْكُمُ'alaykumu ٱلْيَوْمَl-yawma وَلَآwalaa أَنتُمْantum تَحْزَنُونَtahzanoon٦٨ ٱلَّذِينَalladheena ءَامَنُواْaamanooaa بِئَايَٰتِنَاbi'aayaatinaa وَكَانُواْwakaanooaa مُسْلِمِينَmuslimeen٦٩ ٱدْخُلُواْudkhulooaa ٱلْجَنَّةَl-jannaha أَنتُمْantum وَأَزْوَٰجُكُمْwa'azwaajukum تُحْبَرُونَtuhbaroon٧٠ يُطَافُyutaafu عَلَيْهِم'alayhim بِصِحَافٍbisihaafin مِّنmin ذَهَبٍdhahabin وَأَكْوَابٍۖwa'akwaabin وَفِيهَاwafeehaa مَاmaa تَشْتَهِيهِtashtaheehi ٱلْأَنفُسُl-'anfusu وَتَلَذُّwataladhdhu ٱلْأَعْيُنُۖl-'a'yunu وَأَنتُمْwa'antum فِيهَاfeehaa خَٰلِدُونَkhaalidoon٧١ وَتِلْكَwatilka ٱلْجَنَّةُl-jannahu ٱلَّتِيٓl-latee أُورِثْتُمُوهَاoorithtumoohaa بِمَاbimaa كُنتُمْkuntum تَعْمَلُونَta'maloon٧٢ لَكُمْlakum فِيهَاfeehaa فَٰكِهَةٌfaakihahun كَثِيرَةٌkatheerahun مِّنْهَاminhaa تَأْكُلُونَta'kuloon٧٣ إِنَّinna ٱلْمُجْرِمِينَl-mujrimeena فِيfee عَذَابِ'adhaabi جَهَنَّمَjahannama خَٰلِدُونَkhaalidoon٧٤ لَاlaa يُفَتَّرُyufattaru عَنْهُمْ'anhum وَهُمْwahum فِيهِfeehi مُبْلِسُونَmublisoon٧٥ وَمَاwamaa ظَلَمْنَٰهُمْzalamnaahum وَلَٰكِنwalaakin كَانُواْkaanooaa هُمُhumu ٱلظَّٰلِمِينَz-zaalimeen٧٦ وَنَادَوْاْwanaadawaa يَٰمَٰلِكُyaamaaliku لِيَقْضِliyaqdi عَلَيْنَا'alaynaa رَبُّكَۖrabbuka قَالَqaala إِنَّكُمinnakum مَّٰكِثُونَmaakithoon٧٧ لَقَدْlaqad جِئْنَٰكُمji'naakum بِٱلْحَقِّbilhaqqi وَلَٰكِنَّwalaakinna أَكْثَرَكُمْaktharakum لِلْحَقِّlilhaqqi كَٰرِهُونَkaarihoon٧٨ أَمْam أَبْرَمُوٓاْabramooaa أَمْرًاamran فَإِنَّاfa'innaa مُبْرِمُونَmubrimoon٧٩ أَمْam يَحْسَبُونَyahsaboona أَنَّاannaa لَاlaa نَسْمَعُnasma'u سِرَّهُمْsirrahum وَنَجْوَىٰهُمۚwanajwaahum بَلَىٰbalaa وَرُسُلُنَاwarusulunaa لَدَيْهِمْladayhim يَكْتُبُونَyaktuboon٨٠ قُلْqul إِنin كَانَkaana لِلرَّحْمَٰنِlirrahmaani وَلَدٌwaladun فَأَنَا۠fa'anaa أَوَّلُawwalu ٱلْعَٰبِدِينَl-'aabideen٨١ سُبْحَٰنَsubhaana رَبِّrabbi ٱلسَّمَٰوَٰتِs-samaawaati وَٱلْأَرْضِwal'ardi رَبِّrabbi ٱلْعَرْشِl-'arshi عَمَّا'ammaa يَصِفُونَyasifoon٨٢ فَذَرْهُمْfadharhum يَخُوضُواْyakhoodooaa وَيَلْعَبُواْwayal'abooaa حَتَّىٰhattaa يُلَٰقُواْyulaaqooaa يَوْمَهُمُyawmahumu ٱلَّذِيl-ladhee يُوعَدُونَyoo'adoon٨٣ وَهُوَwahuwa ٱلَّذِيl-ladhee فِيfee ٱلسَّمَآءِs-samaa'i إِلَٰهٌilaahun وَفِيwafee ٱلْأَرْضِl-'ardi إِلَٰهٌۚilaahun وَهُوَwahuwa ٱلْحَكِيمُl-hakeemu ٱلْعَلِيمُl-'aleem٨٤ وَتَبَارَكَwatabaaraka ٱلَّذِيl-ladhee لَهُۥlahoo مُلْكُmulku ٱلسَّمَٰوَٰتِs-samaawaati وَٱلْأَرْضِwal'ardi وَمَاwamaa بَيْنَهُمَاbaynahumaa وَعِندَهُۥwa'indahoo عِلْمُ'ilmu ٱلسَّاعَةِs-saa'ahi وَإِلَيْهِwa'ilayhi تُرْجَعُونَturja'oon٨٥ وَلَاwalaa يَمْلِكُyamliku ٱلَّذِينَl-ladheena يَدْعُونَyad'oona مِنmin دُونِهِdoonihi ٱلشَّفَٰعَةَsh-shafaa'aha إِلَّاillaa مَنman شَهِدَshahida بِٱلْحَقِّbilhaqqi وَهُمْwahum يَعْلَمُونَya'lamoon٨٦ وَلَئِنwala'in سَأَلْتَهُمsa'altahum مَّنْman خَلَقَهُمْkhalaqahum لَيَقُولُنَّlayaqoolunna ٱللَّهُۖl-laahu فَأَنَّىٰfa'annaa يُؤْفَكُونَyu'fakoon٨٧ وَقِيلِهِۦwaqeelihee يَٰرَبِّyaarabbi إِنَّinna هَٰٓؤُلَآءِhaa'ulaa'i قَوْمٌqawmun لَّاlaa يُؤْمِنُونَyu'minoon٨٨ فَٱصْفَحْfasfah عَنْهُمْ'anhum وَقُلْwaqul سَلَٰمٌۚsalaamun فَسَوْفَfasawfa يَعْلَمُونَya'lamoon٨٩
