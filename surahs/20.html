وَقُلwaqul رَّبِّrabbi زِدْنِيzidnee عِلْمًا'ilmaa١١٤ وَلَقَدْwalaqad عَهِدْنَآ'ahidnaa إِلَىٰٓilaa ءَادَمَaadama مِنmin قَبْلُqablu فَنَسِيَfanasiya وَلَمْwalam نَجِدْnajid لَهُۥlahoo عَزْمًا'azmaa١١٥ وَإِذْwa'idh قُلْنَاqulnaa لِلْمَلَٰٓئِكَةِlilmalaa'ikahi ٱسْجُدُواْs-judooaa لِأٓدَمَli'dama فَسَجَدُوٓاْfasajadooaa إِلَّآillaa إِبْلِيسَibleesa أَبَىٰabaa١١٦ فَقُلْنَاfaqulnaa يَٰٓئَادَمُyaa'aadamu إِنَّinna هَٰذَاhaadhaa عَدُوٌّ'aduwwun لَّكَlaka وَلِزَوْجِكَwalizawjika فَلَاfalaa يُخْرِجَنَّكُمَاyukhrijannakumaa مِنَmina ٱلْجَنَّةِl-jannahi فَتَشْقَىٰٓfatashqaa١١٧ إِنَّinna لَكَlaka أَلَّاallaa تَجُوعَtajoo'a فِيهَاfeehaa وَلَاwalaa تَعْرَىٰta'raa١١٨ وَأَنَّكَwa'annaka لَاlaa تَظْمَؤُاْtazma'uaa فِيهَاfeehaa وَلَاwalaa تَضْحَىٰtadhaa١١٩ فَوَسْوَسَfawaswasa إِلَيْهِilayhi ٱلشَّيْطَٰنُsh-shaytaanu قَالَqaala يَٰٓئَادَمُyaa'aadamu هَلْhal أَدُلُّكَadulluka عَلَىٰ'alaa شَجَرَةِshajarahi ٱلْخُلْدِl-khuldi وَمُلْكٍwamulkin لَّاlaa يَبْلَىٰyablaa١٢٠ فَأَكَلَاfa'akalaa مِنْهَاminhaa فَبَدَتْfabadat لَهُمَاlahumaa سَوْءَٰتُهُمَاsaw'aatuhumaa وَطَفِقَاwatafiqaa يَخْصِفَانِyakhsifaani عَلَيْهِمَا'alayhimaa مِنmin وَرَقِwaraqi ٱلْجَنَّةِۚl-jannahi وَعَصَىٰٓwa'asaa ءَادَمُaadamu رَبَّهُۥrabbahoo فَغَوَىٰfaghawaa١٢١ ثُمَّthumma ٱجْتَبَٰهُj-tabaahu رَبُّهُۥrabbuhoo فَتَابَfataaba عَلَيْهِ'alayhi وَهَدَىٰwahadaa١٢٢ قَالَqaala ٱهْبِطَاh-bitaa مِنْهَاminhaa جَمِيعَۢاۖjamee'aa بَعْضُكُمْba'dukum لِبَعْضٍliba'din عَدُوٌّۖ'aduwwun فَإِمَّاfa'immaa يَأْتِيَنَّكُمya'tiyannakum مِّنِّيminnee هُدًىhudan فَمَنِfamani ٱتَّبَعَt-taba'a هُدَايَhudaaya فَلَاfalaa يَضِلُّyadillu وَلَاwalaa يَشْقَىٰyashqaa١٢٣ وَمَنْwaman أَعْرَضَa'rada عَن'an ذِكْرِيdhikree فَإِنَّfa'inna لَهُۥlahoo مَعِيشَةًma'eeshahan ضَنكًاdankan وَنَحْشُرُهُۥwanahshuruhoo يَوْمَyawma ٱلْقِيَٰمَةِl-qiyaamahi أَعْمَىٰa'maa١٢٤ قَالَqaala رَبِّrabbi لِمَlima حَشَرْتَنِيٓhashartanee أَعْمَىٰa'maa وَقَدْwaqad كُنتُkuntu بَصِيرًاbaseeraa١٢٥ قَالَqaala كَذَٰلِكَkadhaalika أَتَتْكَatatka ءَايَٰتُنَاaayaatunaa فَنَسِيتَهَاۖfanaseetahaa وَكَذَٰلِكَwakadhaalika ٱلْيَوْمَl-yawma تُنسَىٰtunsaa١٢٦ وَكَذَٰلِكَwakadhaalika نَجْزِيnajzee مَنْman أَسْرَفَasrafa وَلَمْwalam يُؤْمِنۢyu'min بِئَايَٰتِbi'aayaati رَبِّهِۦۚrabbihee وَلَعَذَابُwala'adhaabu ٱلْأٓخِرَةِl-'khirahi أَشَدُّashaddu وَأَبْقَىٰٓwa'abqaa١٢٧ أَفَلَمْafalam يَهْدِyahdi لَهُمْlahum كَمْkam أَهْلَكْنَاahlaknaa قَبْلَهُمqablahum مِّنَmina ٱلْقُرُونِl-qurooni يَمْشُونَyamshoona فِيfee مَسَٰكِنِهِمْۚmasaakinihim إِنَّinna فِيfee ذَٰلِكَdhaalika لَأٓيَٰتٍla'yaatin لِّأُوْلِيli'oolee ٱلنُّهَىٰn-nuhaa١٢٨ وَلَوْلَاwalawlaa كَلِمَةٌkalimahun سَبَقَتْsabaqat مِنmin رَّبِّكَrabbika لَكَانَlakaana لِزَامًاlizaaman وَأَجَلٌwa'ajalun مُّسَمًّىmusammaa١٢٩ فَٱصْبِرْfasbir عَلَىٰ'alaa مَاmaa يَقُولُونَyaqooloona وَسَبِّحْwasabbih بِحَمْدِbihamdi رَبِّكَrabbika قَبْلَqabla طُلُوعِtuloo'i ٱلشَّمْسِsh-shamsi وَقَبْلَwaqabla غُرُوبِهَاۖghuroobihaa وَمِنْwamin ءَانَآيِٕaanaayi ٱلَّيْلِl-layli فَسَبِّحْfasabbih وَأَطْرَافَwa'atraafa ٱلنَّهَارِn-nahaari لَعَلَّكَla'allaka تَرْضَىٰtardaa١٣٠ وَلَاwalaa تَمُدَّنَّtamuddanna عَيْنَيْكَ'aynayka إِلَىٰilaa مَاmaa مَتَّعْنَاmatta'naa بِهِۦٓbihee أَزْوَٰجًاazwaajan مِّنْهُمْminhum زَهْرَةَzahraha ٱلْحَيَوٰةِl-hayahi ٱلدُّنْيَاd-dunyaa لِنَفْتِنَهُمْlinaftinahum فِيهِۚfeehi وَرِزْقُwarizqu رَبِّكَrabbika خَيْرٌkhayrun وَأَبْقَىٰwa'abqaa١٣١ وَأْمُرْwa'mur أَهْلَكَahlaka بِٱلصَّلَوٰةِbissalahi وَٱصْطَبِرْwastabir عَلَيْهَاۖ'alayhaa لَاlaa نَسْئَلُكَnas'aluka رِزْقًاۖrizqan نَّحْنُnahnu نَرْزُقُكَۗnarzuquka وَٱلْعَٰقِبَةُwal'aaqibahu لِلتَّقْوَىٰlittaqwaa١٣٢ وَقَالُواْwaqaalooaa لَوْلَاlawlaa يَأْتِينَاya'teenaa بِئَايَةٍbi'aayahin مِّنmin رَّبِّهِۦٓۚrabbihee أَوَلَمْawalam تَأْتِهِمta'tihim بَيِّنَةُbayyinahu مَاmaa فِيfee ٱلصُّحُفِs-suhufi ٱلْأُولَىٰl-'oolaa١٣٣ وَلَوْwalaw أَنَّآannaa أَهْلَكْنَٰهُمahlaknaahum بِعَذَابٍbi'adhaabin مِّنmin قَبْلِهِۦqablihee لَقَالُواْlaqaalooaa رَبَّنَاrabbanaa لَوْلَآlawlaa أَرْسَلْتَarsalta إِلَيْنَاilaynaa رَسُولًاrasoolan فَنَتَّبِعَfanattabi'a ءَايَٰتِكَaayaatika مِنmin قَبْلِqabli أَنan نَّذِلَّnadhilla وَنَخْزَىٰwanakhzaa١٣٤ قُلْqul كُلٌّkullun مُّتَرَبِّصٌmutarabbisun فَتَرَبَّصُواْۖfatarabbasooaa فَسَتَعْلَمُونَfasata'lamoona مَنْman أَصْحَٰبُashaabu ٱلصِّرَٰطِs-siraati ٱلسَّوِيِّs-sawiyyi وَمَنِwamani ٱهْتَدَىٰh-tadaa١٣٥
