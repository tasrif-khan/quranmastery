  • Madd al-Aaridمد عارضHold 2, 4 or 6 counts when stopping.

Al-Ghashiyah الغاشية Meccan · 26 ayat
هَلْhal أَتَىٰكَataaka حَدِيثُhadeethu ٱلْغَٰشِيَةِl-ghaashiyah١ وُجُوهٌwujoohun يَوْمَئِذٍyawma'idhin خَٰشِعَةٌkhaashi'ah٢ عَامِلَةٌ'aamilahun نَّاصِبَةٌnaasibah٣ تَصْلَىٰtaslaa نَارًاnaaran حَامِيَةًhaamiyahaa٤ تُسْقَىٰtusqaa مِنْmin عَيْنٍ'aynin ءَانِيَةٍaaniyah٥ لَّيْسَllaysa لَهُمْlahum طَعَامٌta'aamun إِلَّاillaa مِنmin ضَرِيعٍdaree'٦ لَّاllaa يُسْمِنُyusminu وَلَاwalaa يُغْنِيyughnee مِنmin جُوعٍjoo'٧ وُجُوهٌwujoohun يَوْمَئِذٍyawma'idhin نَّاعِمَةٌnaa'imah٨ لِّسَعْيِهَاllisa'yihaa رَاضِيَةٌraadiyah٩ فِيfee جَنَّةٍjannahin عَالِيَةٍ'aaliyah١٠ لَّاllaa تَسْمَعُtasma'u فِيهَاfeehaa لَٰغِيَةًlaaghiyahaa١١ فِيهَاfeehaa عَيْنٌ'aynun جَارِيَةٌjaariyah١٢ فِيهَاfeehaa سُرُرٌsururun مَّرْفُوعَةٌmarfoo'ah١٣ وَأَكْوَابٌwa'akwaabun مَّوْضُوعَةٌmawdoo'ah١٤ وَنَمَارِقُwanamaariqu مَصْفُوفَةٌmasfoofah١٥ وَزَرَابِيُّwazaraabiyyu مَبْثُوثَةٌmabthoothah١٦ أَفَلَاafalaa يَنظُرُونَyanzuroona إِلَىilaa ٱلْإِبِلِl-'ibili كَيْفَkayfa خُلِقَتْkhuliqat١٧ وَإِلَىwa'ilaa ٱلسَّمَآءِs-samaa'i كَيْفَkayfa رُفِعَتْrufi'at١٨ وَإِلَىwa'ilaa ٱلْجِبَالِl-jibaali كَيْفَkayfa نُصِبَتْnusibat١٩ وَإِلَىwa'ilaa ٱلْأَرْضِl-'ardi كَيْفَkayfa سُطِحَتْsutihat٢٠ فَذَكِّرْfadhakkir إِنَّمَآinnamaa أَنتَanta مُذَكِّرٌmudhakkir٢١ لَّسْتَllasta عَلَيْهِم'alayhim بِمُصَيْطِرٍbimusaytir٢٢ إِلَّاillaa مَنman تَوَلَّىٰtawallaa وَكَفَرَwakafar٢٣ فَيُعَذِّبُهُfayu'adhdhibuhu ٱللَّهُl-laahu ٱلْعَذَابَl-'adhaaba ٱلْأَكْبَرَl-'akbar٢٤ إِنَّinna إِلَيْنَآilaynaa إِيَابَهُمْiyaabahum٢٥ ثُمَّthumma إِنَّinna عَلَيْنَا'alaynaa حِسَابَهُمhisaabahum٢٦
