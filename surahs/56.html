Tajweed keyrules in this surah
  • Madd Laazimمد لازمHold 6 counts. Obligatory.
  • Madd Muttasilمد متصلHold 4 to 5 counts.
  • Madd Munfasilمد منفصلHold 4 to 5 counts.
  • Madd al-Aaridمد عارضHold 2, 4 or 6 counts when stopping.
  • IqlabإقلابNoon turns into a meem sound.
  • Idghaam w/ Ghunnahإدغام بغنةMerge into the next letter with nasalisation.
  • Idghaam w/o Ghunnahإدغام بغير غنةMerge into the next letter, no nasalisation.
  • Idghaam Shafawiإدغام شفويMeem merges into meem.
  • IkhfaإخفاءHide the noon with a light nasal sound.
  • GhunnahغنةNasal sound, 2 counts.
  • QalqalahقلقلةEcho the letter with a slight bounce.
  • Lam Shamsiyyahلام شمسيةSilent laam, merged into the next letter.
  • Hamzat al-Waslهمزة وصلSkipped when joined to the word before.
  • Madd Tabee'iمد طبيعيNatural madd. Hold 2 counts.
Al-Waqi'ah الواقعة Meccan · 96 ayat
إِذَاidhaa وَقَعَتِwaqa'ati ٱلْوَاقِعَةُl-waaqi'ah١ لَيْسَlaysa لِوَقْعَتِهَاliwaq'atihaa كَاذِبَةٌkaadhibah٢ خَافِضَةٌkhaafidahun رَّافِعَةٌraafi'ah٣ إِذَاidhaa رُجَّتِrujjati ٱلْأَرْضُl-'ardu رَجًّاrajjaa٤ وَبُسَّتِwabussati ٱلْجِبَالُl-jibaalu بَسًّاbassaa٥ فَكَانَتْfakaanat هَبَآءًhabaa'an مُّنۢبَثًّاmunbaththaa٦ وَكُنتُمْwakuntum أَزْوَٰجًاazwaajan ثَلَٰثَةًthalaathahaa٧ فَأَصْحَٰبُfa'ashaabu ٱلْمَيْمَنَةِl-maymanahi مَآmaa أَصْحَٰبُashaabu ٱلْمَيْمَنَةِl-maymanah٨ وَأَصْحَٰبُwa'ashaabu ٱلْمَشْئَمَةِl-mash'amahi مَآmaa أَصْحَٰبُashaabu ٱلْمَشْئَمَةِl-mash'amah٩ وَٱلسَّٰبِقُونَwassaabiqoona ٱلسَّٰبِقُونَs-saabiqoon١٠ أُوْلَٰٓئِكَoolaa'ika ٱلْمُقَرَّبُونَl-muqarraboon١١ فِيfee جَنَّٰتِjannaati ٱلنَّعِيمِn-na'eem١٢ ثُلَّةٌthullahun مِّنَmina ٱلْأَوَّلِينَl-'awwaleen١٣ وَقَلِيلٌwaqaleelun مِّنَmina ٱلْأٓخِرِينَl-'khireen١٤ عَلَىٰ'alaa سُرُرٍsururin مَّوْضُونَةٍmawdoonah١٥ مُّتَّكِئِينَmmuttaki'eena عَلَيْهَا'alayhaa مُتَقَٰبِلِينَmutaqaabileen١٦ يَطُوفُyatoofu عَلَيْهِمْ'alayhim وِلْدَٰنٌwildaanun مُّخَلَّدُونَmukhalladoon١٧ بِأَكْوَابٍbi'akwaabin وَأَبَارِيقَwa'abaareeqa وَكَأْسٍwaka'sin مِّنmin مَّعِينٍma'een١٨ لَّاllaa يُصَدَّعُونَyusadda'oona عَنْهَا'anhaa وَلَاwalaa يُنزِفُونَyunzifoon١٩ وَفَٰكِهَةٍwafaakihahin مِّمَّاmimmaa يَتَخَيَّرُونَyatakhayyaroon٢٠ وَلَحْمِwalahmi طَيْرٍtayrin مِّمَّاmimmaa يَشْتَهُونَyashtahoon٢١ وَحُورٌwahoorun عِينٌ'een٢٢ كَأَمْثَٰلِka'amthaali ٱللُّؤْلُوِٕl-lu'looi ٱلْمَكْنُونِl-maknoon٢٣ جَزَآءَۢjazaa'a بِمَاbimaa كَانُواْkaanooaa يَعْمَلُونَya'maloon٢٤ لَاlaa يَسْمَعُونَyasma'oona فِيهَاfeehaa لَغْوًاlaghwan وَلَاwalaa تَأْثِيمًاta'theemaa٢٥ إِلَّاillaa قِيلًاqeelan سَلَٰمًاsalaaman سَلَٰمًاsalaamaa٢٦ وَأَصْحَٰبُwa'ashaabu ٱلْيَمِينِl-yameeni مَآmaa أَصْحَٰبُashaabu ٱلْيَمِينِl-yameen٢٧ فِيfee سِدْرٍsidrin مَّخْضُودٍmakhdood٢٨ وَطَلْحٍwatalhin مَّنضُودٍmandood٢٩ وَظِلٍّwazillin مَّمْدُودٍmamdood٣٠ وَمَآءٍwamaa'in مَّسْكُوبٍmaskoob٣١ وَفَٰكِهَةٍwafaakihahin كَثِيرَةٍkatheerah٣٢ لَّاllaa مَقْطُوعَةٍmaqtoo'ahin وَلَاwalaa مَمْنُوعَةٍmamnoo'ah٣٣ وَفُرُشٍwafurushin مَّرْفُوعَةٍmarfoo'ah٣٤ إِنَّآinnaa أَنشَأْنَٰهُنَّansha'naahunna إِنشَآءًinshaa'aa٣٥ فَجَعَلْنَٰهُنَّfaja'alnaahunna أَبْكَارًاabkaaraa٣٦ عُرُبًا'uruban أَتْرَابًاatraabaa٣٧ لِّأَصْحَٰبِlli'ashaabi ٱلْيَمِينِl-yameen٣٨ ثُلَّةٌthullahun مِّنَmina ٱلْأَوَّلِينَl-'awwaleen٣٩ وَثُلَّةٌwathullahun مِّنَmina ٱلْأٓخِرِينَl-'khireen٤٠ وَأَصْحَٰبُwa'ashaabu ٱلشِّمَالِsh-shimaali مَآmaa أَصْحَٰبُashaabu ٱلشِّمَالِsh-shimaal٤١ فِيfee سَمُومٍsamoomin وَحَمِيمٍwahameem٤٢ وَظِلٍّwazillin مِّنmin يَحْمُومٍyahmoom٤٣ لَّاllaa بَارِدٍbaaridin وَلَاwalaa كَرِيمٍkareem٤٤ إِنَّهُمْinnahum كَانُواْkaanooaa قَبْلَqabla ذَٰلِكَdhaalika مُتْرَفِينَmutrafeen٤٥ وَكَانُواْwakaanooaa يُصِرُّونَyusirroona عَلَى'alaa ٱلْحِنثِl-hinthi ٱلْعَظِيمِl-'azeem٤٦ وَكَانُواْwakaanooaa يَقُولُونَyaqooloona أَئِذَاa'idhaa مِتْنَاmitnaa وَكُنَّاwakunnaa تُرَابًاturaaban وَعِظَٰمًاwa'izaaman أَءِنَّاa'innaa لَمَبْعُوثُونَlamab'oothoon٤٧ أَوَءَابَآؤُنَاawa'aabaa'unaa ٱلْأَوَّلُونَl-'awwaloon٤٨ قُلْqul إِنَّinna ٱلْأَوَّلِينَl-'awwaleena وَٱلْأٓخِرِينَwal'khireen٤٩ لَمَجْمُوعُونَlamajmoo'oona إِلَىٰilaa مِيقَٰتِmeeqaati يَوْمٍyawmin مَّعْلُومٍma'loom٥٠ ثُمَّthumma إِنَّكُمْinnakum أَيُّهَاayyuhaa ٱلضَّآلُّونَd-daalloona ٱلْمُكَذِّبُونَl-mukadhdhiboon٥١ لَأٓكِلُونَla'kiloona مِنmin شَجَرٍshajarin مِّنmin زَقُّومٍzaqqoom٥٢ فَمَالِئُونَfamaali'oona مِنْهَاminhaa ٱلْبُطُونَl-butoon٥٣ فَشَٰرِبُونَfashaariboona عَلَيْهِ'alayhi مِنَmina ٱلْحَمِيمِl-hameem٥٤ فَشَٰرِبُونَfashaariboona شُرْبَshurba ٱلْهِيمِl-heem٥٥ هَٰذَاhaadhaa نُزُلُهُمْnuzuluhum يَوْمَyawma ٱلدِّينِd-deen٥٦ نَحْنُnahnu خَلَقْنَٰكُمْkhalaqnaakum فَلَوْلَاfalawlaa تُصَدِّقُونَtusaddiqoon٥٧ أَفَرَءَيْتُمafara'aytum مَّاmaa تُمْنُونَtumnoon٥٨ ءَأَنتُمْa'antum تَخْلُقُونَهُۥٓtakhluqoonahoo أَمْam نَحْنُnahnu ٱلْخَٰلِقُونَl-khaaliqoon٥٩ نَحْنُnahnu قَدَّرْنَاqaddarnaa بَيْنَكُمُbaynakumu ٱلْمَوْتَl-mawta وَمَاwamaa نَحْنُnahnu بِمَسْبُوقِينَbimasbooqeen٦٠ عَلَىٰٓ'alaa أَنan نُّبَدِّلَnubaddila أَمْثَٰلَكُمْamthaalakum وَنُنشِئَكُمْwanunshi'akum فِيfee مَاmaa لَاlaa تَعْلَمُونَta'lamoon٦١ وَلَقَدْwalaqad عَلِمْتُمُ'alimtumu ٱلنَّشْأَةَn-nash'aha ٱلْأُولَىٰl-'oolaa فَلَوْلَاfalawlaa تَذَكَّرُونَtadhakkaroon٦٢ أَفَرَءَيْتُمafara'aytum مَّاmaa تَحْرُثُونَtahruthoon٦٣ ءَأَنتُمْa'antum تَزْرَعُونَهُۥٓtazra'oonahoo أَمْam نَحْنُnahnu ٱلزَّٰرِعُونَz-zaari'oon٦٤ لَوْlaw نَشَآءُnashaa'u لَجَعَلْنَٰهُlaja'alnaahu حُطَٰمًاhutaaman فَظَلْتُمْfazaltum تَفَكَّهُونَtafakkahoon٦٥ إِنَّاinnaa لَمُغْرَمُونَlamughramoon٦٦ بَلْbal نَحْنُnahnu مَحْرُومُونَmahroomoon٦٧ أَفَرَءَيْتُمُafara'aytumu ٱلْمَآءَl-maa'a ٱلَّذِيl-ladhee تَشْرَبُونَtashraboon٦٨ ءَأَنتُمْa'antum أَنزَلْتُمُوهُanzaltumoohu مِنَmina ٱلْمُزْنِl-muzni أَمْam نَحْنُnahnu ٱلْمُنزِلُونَl-munziloon٦٩ لَوْlaw نَشَآءُnashaa'u جَعَلْنَٰهُja'alnaahu أُجَاجًاujaajan فَلَوْلَاfalawlaa تَشْكُرُونَtashkuroon٧٠ أَفَرَءَيْتُمُafara'aytumu ٱلنَّارَn-naara ٱلَّتِيl-latee تُورُونَtooroon٧١ ءَأَنتُمْa'antum أَنشَأْتُمْansha'tum شَجَرَتَهَآshajaratahaa أَمْam نَحْنُnahnu ٱلْمُنشِئُونَl-munshi'oon٧٢ نَحْنُnahnu جَعَلْنَٰهَاja'alnaahaa تَذْكِرَةًtadhkirahan وَمَتَٰعًاwamataa'an لِّلْمُقْوِينَlilmuqween٧٣ فَسَبِّحْfasabbih بِٱسْمِbismi رَبِّكَrabbika ٱلْعَظِيمِl-'azeem٧٤ ۞فَلَآfalaa أُقْسِمُuqsimu بِمَوَٰقِعِbimawaaqi'i ٱلنُّجُومِn-nujoom٧٥ وَإِنَّهُۥwa'innahoo لَقَسَمٌlaqasamun لَّوْlaw تَعْلَمُونَta'lamoona عَظِيمٌ'azeem٧٦ إِنَّهُۥinnahoo لَقُرْءَانٌlaqur'aanun كَرِيمٌkareem٧٧ فِيfee كِتَٰبٍkitaabin مَّكْنُونٍmaknoon٧٨ لَّاllaa يَمَسُّهُۥٓyamassuhoo إِلَّاillaa ٱلْمُطَهَّرُونَl-mutahharoon٧٩ تَنزِيلٌtanzeelun مِّنmin رَّبِّrabbi ٱلْعَٰلَمِينَl-'aalameen٨٠ أَفَبِهَٰذَاafabihaadhaa ٱلْحَدِيثِl-hadeethi أَنتُمantum مُّدْهِنُونَmudhinoon٨١ وَتَجْعَلُونَwataj'aloona رِزْقَكُمْrizqakum أَنَّكُمْannakum تُكَذِّبُونَtukadhdhiboon٨٢ فَلَوْلَآfalawlaa إِذَاidhaa بَلَغَتِbalaghati ٱلْحُلْقُومَl-hulqoom٨٣ وَأَنتُمْwa'antum حِينَئِذٍheena'idhin تَنظُرُونَtanzuroon٨٤ وَنَحْنُwanahnu أَقْرَبُaqrabu إِلَيْهِilayhi مِنكُمْminkum وَلَٰكِنwalaakin لَّاlaa تُبْصِرُونَtubsiroon٨٥ فَلَوْلَآfalawlaa إِنin كُنتُمْkuntum غَيْرَghayra مَدِينِينَmadeeneen٨٦ تَرْجِعُونَهَآtarji'oonahaa إِنin كُنتُمْkuntum صَٰدِقِينَsaadiqeen٨٧ فَأَمَّآfa'ammaa إِنin كَانَkaana مِنَmina ٱلْمُقَرَّبِينَl-muqarrabeen٨٨ فَرَوْحٌfarawhun وَرَيْحَانٌwarayhaanun وَجَنَّتُwajannatu نَعِيمٍna'eem٨٩ وَأَمَّآwa'ammaa إِنin كَانَkaana مِنْmin أَصْحَٰبِashaabi ٱلْيَمِينِl-yameen٩٠ فَسَلَٰمٌfasalaamun لَّكَlaka مِنْmin أَصْحَٰبِashaabi ٱلْيَمِينِl-yameen٩١ وَأَمَّآwa'ammaa إِنin كَانَkaana مِنَmina ٱلْمُكَذِّبِينَl-mukadhdhibeena ٱلضَّآلِّينَd-daalleen٩٢ فَنُزُلٌfanuzulun مِّنْmin حَمِيمٍhameem٩٣ وَتَصْلِيَةُwatasliyahu جَحِيمٍjaheem٩٤ إِنَّinna هَٰذَاhaadhaa لَهُوَlahuwa حَقُّhaqqu ٱلْيَقِينِl-yaqeen٩٥ فَسَبِّحْfasabbih بِٱسْمِbismi رَبِّكَrabbika ٱلْعَظِيمِl-'azeem٩٦
