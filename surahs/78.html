  • Madd al-Aaridمد عارضHold 2, 4 or 6 counts when stopping.

An-Naba النبإ Meccan · 40 ayat
عَمَّ'amma يَتَسَآءَلُونَyatasaa'aloon١ عَنِ'ani ٱلنَّبَإِn-naba'i ٱلْعَظِيمِl-'azeem٢ ٱلَّذِيalladhee هُمْhum فِيهِfeehi مُخْتَلِفُونَmukhtalifoon٣ كَلَّاkallaa سَيَعْلَمُونَsaya'lamoon٤ ثُمَّthumma كَلَّاkallaa سَيَعْلَمُونَsaya'lamoon٥ أَلَمْalam نَجْعَلِnaj'ali ٱلْأَرْضَl-'arda مِهَٰدًاmihaadaa٦ وَٱلْجِبَالَwaljibaala أَوْتَادًاawtaadaa٧ وَخَلَقْنَٰكُمْwakhalaqnaakum أَزْوَٰجًاazwaajaa٨ وَجَعَلْنَاwaja'alnaa نَوْمَكُمْnawmakum سُبَاتًاsubaataa٩ وَجَعَلْنَاwaja'alnaa ٱلَّيْلَl-layla لِبَاسًاlibaasaa١٠ وَجَعَلْنَاwaja'alnaa ٱلنَّهَارَn-nahaara مَعَاشًاma'aashaa١١ وَبَنَيْنَاwabanaynaa فَوْقَكُمْfawqakum سَبْعًاsab'an شِدَادًاshidaadaa١٢ وَجَعَلْنَاwaja'alnaa سِرَاجًاsiraajan وَهَّاجًاwahhaajaa١٣ وَأَنزَلْنَاwa'anzalnaa مِنَmina ٱلْمُعْصِرَٰتِl-mu'siraati مَآءًmaa'an ثَجَّاجًاthajjaajaa١٤ لِّنُخْرِجَllinukhrija بِهِۦbihee حَبًّاhabban وَنَبَاتًاwanabaataa١٥ وَجَنَّٰتٍwajannaatin أَلْفَافًاalfaafaa١٦ إِنَّinna يَوْمَyawma ٱلْفَصْلِl-fasli كَانَkaana مِيقَٰتًاmeeqaataa١٧ يَوْمَyawma يُنفَخُyunfakhu فِيfee ٱلصُّورِs-soori فَتَأْتُونَfata'toona أَفْوَاجًاafwaajaa١٨ وَفُتِحَتِwafutihati ٱلسَّمَآءُs-samaa'u فَكَانَتْfakaanat أَبْوَٰبًاabwaabaa١٩ وَسُيِّرَتِwasuyyirati ٱلْجِبَالُl-jibaalu فَكَانَتْfakaanat سَرَابًاsaraabaa٢٠ إِنَّinna جَهَنَّمَjahannama كَانَتْkaanat مِرْصَادًاmirsaadaa٢١ لِّلطَّٰغِينَllittaagheena مَئَابًاma'aabaa٢٢ لَّٰبِثِينَllaabitheena فِيهَآfeehaa أَحْقَابًاahqaabaa٢٣ لَّاllaa يَذُوقُونَyadhooqoona فِيهَاfeehaa بَرْدًاbardan وَلَاwalaa شَرَابًاsharaabaa٢٤ إِلَّاillaa حَمِيمًاhameeman وَغَسَّاقًاwaghassaaqaa٢٥ جَزَآءًjazaa'an وِفَاقًاwifaaqaa٢٦ إِنَّهُمْinnahum كَانُواْkaanooaa لَاlaa يَرْجُونَyarjoona حِسَابًاhisaabaa٢٧ وَكَذَّبُواْwakadhdhabooaa بِئَايَٰتِنَاbi'aayaatinaa كِذَّابًاkidhdhaabaa٢٨ وَكُلَّwakulla شَيْءٍshay'in أَحْصَيْنَٰهُahsaynaahu كِتَٰبًاkitaabaa٢٩ فَذُوقُواْfadhooqooaa فَلَنfalan نَّزِيدَكُمْnazeedakum إِلَّاillaa عَذَابًا'adhaabaa٣٠ إِنَّinna لِلْمُتَّقِينَlilmuttaqeena مَفَازًاmafaazaa٣١ حَدَآئِقَhadaa'iqa وَأَعْنَٰبًاwa'a'naabaa٣٢ وَكَوَاعِبَwakawaa'iba أَتْرَابًاatraabaa٣٣ وَكَأْسًاwaka'san دِهَاقًاdihaaqaa٣٤ لَّاllaa يَسْمَعُونَyasma'oona فِيهَاfeehaa لَغْوًاlaghwan وَلَاwalaa كِذَّٰبًاkidhdhaabaa٣٥ جَزَآءًjazaa'an مِّنmin رَّبِّكَrabbika عَطَآءً'ataa'an حِسَابًاhisaabaa٣٦ رَّبِّrrabbi ٱلسَّمَٰوَٰتِs-samaawaati وَٱلْأَرْضِwal'ardi وَمَاwamaa بَيْنَهُمَاbaynahumaa ٱلرَّحْمَٰنِۖr-rahmaani لَاlaa يَمْلِكُونَyamlikoona مِنْهُminhu خِطَابًاkhitaabaa٣٧ يَوْمَyawma يَقُومُyaqoomu ٱلرُّوحُr-roohu وَٱلْمَلَٰٓئِكَةُwalmalaa'ikahu صَفًّاۖsaffan لَّاlaa يَتَكَلَّمُونَyatakallamoona إِلَّاillaa مَنْman أَذِنَadhina لَهُlahu ٱلرَّحْمَٰنُr-rahmaanu وَقَالَwaqaala صَوَابًاsawaabaa٣٨ ذَٰلِكَdhaalika ٱلْيَوْمُl-yawmu ٱلْحَقُّۖl-haqqu فَمَنfaman شَآءَshaa'a ٱتَّخَذَt-takhadha إِلَىٰilaa رَبِّهِۦrabbihee مَئَابًاma'aabaa٣٩ إِنَّآinnaa أَنذَرْنَٰكُمْandharnaakum عَذَابًا'adhaaban قَرِيبًاqareeban يَوْمَyawma يَنظُرُyanzuru ٱلْمَرْءُl-mar'u مَاmaa قَدَّمَتْqaddamat يَدَاهُyadaahu وَيَقُولُwayaqoolu ٱلْكَافِرُl-kaafiru يَٰلَيْتَنِيyaalaytanee كُنتُkuntu تُرَٰبَۢاturaabaa٤٠
