Al-Muzzammil المزمل Meccan · 20 ayat
يَٰٓأَيُّهَاyaa'ayyuhaa ٱلْمُزَّمِّلُl-muzzammil١ قُمِqumi ٱلَّيْلَl-layla إِلَّاillaa قَلِيلًاqaleelaa٢ نِّصْفَهُۥٓnnisfahoo أَوِawi ٱنقُصْn-qus مِنْهُminhu قَلِيلًاqaleelaa٣ أَوْaw زِدْzid عَلَيْهِ'alayhi وَرَتِّلِwarattili ٱلْقُرْءَانَl-qur'aana تَرْتِيلًاtarteelaa٤ إِنَّاinnaa سَنُلْقِيsanulqee عَلَيْكَ'alayka قَوْلًاqawlan ثَقِيلًاthaqeelaa٥ إِنَّinna نَاشِئَةَnaashi'aha ٱلَّيْلِl-layli هِيَhiya أَشَدُّashaddu وَطْئًاwat'an وَأَقْوَمُwa'aqwamu قِيلًاqeelaa٦ إِنَّinna لَكَlaka فِيfee ٱلنَّهَارِn-nahaari سَبْحًاsabhan طَوِيلًاtaweelaa٧ وَٱذْكُرِwadhkuri ٱسْمَs-ma رَبِّكَrabbika وَتَبَتَّلْwatabattal إِلَيْهِilayhi تَبْتِيلًاtabteelaa٨ رَّبُّrrabbu ٱلْمَشْرِقِl-mashriqi وَٱلْمَغْرِبِwalmaghribi لَآlaa إِلَٰهَilaaha إِلَّاillaa هُوَhuwa فَٱتَّخِذْهُfattakhidhhu وَكِيلًاwakeelaa٩ وَٱصْبِرْwasbir عَلَىٰ'alaa مَاmaa يَقُولُونَyaqooloona وَٱهْجُرْهُمْwahjurhum هَجْرًاhajran جَمِيلًاjameelaa١٠ وَذَرْنِيwadharnee وَٱلْمُكَذِّبِينَwalmukadhdhibeena أُوْلِيoolee ٱلنَّعْمَةِn-na'mahi وَمَهِّلْهُمْwamahhilhum قَلِيلًاqaleelaa١١ إِنَّinna لَدَيْنَآladaynaa أَنكَالًاankaalan وَجَحِيمًاwajaheemaa١٢ وَطَعَامًاwata'aaman ذَاdhaa غُصَّةٍghussahin وَعَذَابًاwa'adhaaban أَلِيمًاaleemaa١٣ يَوْمَyawma تَرْجُفُtarjufu ٱلْأَرْضُl-'ardu وَٱلْجِبَالُwaljibaalu وَكَانَتِwakaanati ٱلْجِبَالُl-jibaalu كَثِيبًاkatheeban مَّهِيلًاmaheelaa١٤ إِنَّآinnaa أَرْسَلْنَآarsalnaa إِلَيْكُمْilaykum رَسُولًاrasoolan شَٰهِدًاshaahidan عَلَيْكُمْ'alaykum كَمَآkamaa أَرْسَلْنَآarsalnaa إِلَىٰilaa فِرْعَوْنَfir'awna رَسُولًاrasoolaa١٥ فَعَصَىٰfa'asaa فِرْعَوْنُfir'awnu ٱلرَّسُولَr-rasoola فَأَخَذْنَٰهُfa'akhadhnaahu أَخْذًاakhdhan وَبِيلًاwabeelaa١٦ فَكَيْفَfakayfa تَتَّقُونَtattaqoona إِنin كَفَرْتُمْkafartum يَوْمًاyawman يَجْعَلُyaj'alu ٱلْوِلْدَٰنَl-wildaana شِيبًاsheebaa١٧ ٱلسَّمَآءُassamaa'u مُنفَطِرُۢmunfatiru بِهِۦۚbihee كَانَkaana وَعْدُهُۥwa'duhoo مَفْعُولًاmaf'oolaa١٨ إِنَّinna هَٰذِهِۦhaadhihee تَذْكِرَةٌۖtadhkirahun فَمَنfaman شَآءَshaa'a ٱتَّخَذَt-takhadha إِلَىٰilaa رَبِّهِۦrabbihee سَبِيلًاsabeelaa١٩ ۞إِنَّ'inna رَبَّكَrabbaka يَعْلَمُya'lamu أَنَّكَannaka تَقُومُtaqoomu أَدْنَىٰadnaa مِنmin ثُلُثَيِthuluthayi ٱلَّيْلِl-layli وَنِصْفَهُۥwanisfahoo وَثُلُثَهُۥwathuluthahoo وَطَآئِفَةٌwataa'ifahun مِّنَmina ٱلَّذِينَl-ladheena مَعَكَۚma'aka وَٱللَّهُwalllaahu يُقَدِّرُyuqaddiru ٱلَّيْلَl-layla وَٱلنَّهَارَۚwalnnahaara عَلِمَ'alima أَنan لَّنlan تُحْصُوهُtuhsoohu فَتَابَfataaba عَلَيْكُمْۖ'alaykum فَٱقْرَءُواْfaqra'ooaa مَاmaa تَيَسَّرَtayassara مِنَmina ٱلْقُرْءَانِۚl-qur'aani عَلِمَ'alima أَنan سَيَكُونُsayakoonu مِنكُمminkum مَّرْضَىٰmardaa وَءَاخَرُونَwa'aakharoona يَضْرِبُونَyadriboona فِيfee ٱلْأَرْضِl-'ardi يَبْتَغُونَyabtaghoona مِنmin فَضْلِfadli ٱللَّهِl-llaahi وَءَاخَرُونَwa'aakharoona يُقَٰتِلُونَyuqaatiloona فِيfee سَبِيلِsabeeli ٱللَّهِۖl-llaahi فَٱقْرَءُواْfaqra'ooaa مَاmaa تَيَسَّرَtayassara مِنْهُۚminhu وَأَقِيمُواْwa'aqeemooaa ٱلصَّلَوٰةَl-ssalawaaha وَءَاتُواْwa'aatooaa ٱلزَّكَوٰةَl-zzakawaaha وَأَقْرِضُواْwa'aqridooaa ٱللَّهَl-llaaha قَرْضًاqardan حَسَنًاۚhasanan وَمَاwamaa تُقَدِّمُواْtuqaddimooaa لِأَنفُسِكُمli'anfusikum مِّنْmin خَيْرٍkhayrin تَجِدُوهُtajidoohu عِندَ'inda ٱللَّهِl-llaahi هُوَhuwa خَيْرًاkhayran وَأَعْظَمَwa'a'zama أَجْرًاۚajran
