  • Madd Muttasilمد متصلHold 4 to 5 counts.

  • Idghaam w/o Ghunnahإدغام بغير غنةMerge into the next letter, no nasalisation.
  • Idghaam Shafawiإدغام شفويMeem merges into meem.

  • Idghaam Mutaqaribaynإدغام متقاربينLetters of close articulation merge.

لَهُۥlahoo طَلَبًاtalabaa٤١ وَأُحِيطَwa'uheeta بِثَمَرِهِۦbithamarihee فَأَصْبَحَfa'asbaha يُقَلِّبُyuqallibu كَفَّيْهِkaffayhi عَلَىٰ'alaa مَآmaa أَنفَقَanfaqa فِيهَاfeehaa وَهِيَwahiya خَاوِيَةٌkhaawiyahun عَلَىٰ'alaa عُرُوشِهَا'urooshihaa وَيَقُولُwayaqoolu يَٰلَيْتَنِيyaalaytanee لَمْlam أُشْرِكْushrik بِرَبِّيٓbirabbee أَحَدًاahadaa٤٢ وَلَمْwalam تَكُنtakun لَّهُۥlahoo هُوَhuwa خَيْرٌkhayrun ثَوَابًاthawaaban وَخَيْرٌwakhayrun عُقْبًا'uqbaa٤٤ وَٱضْرِبْwadrib لَهُمlahum مَّثَلَmathala ٱلْحَيَوٰةِl-hayahi ٱلدُّنْيَاd-dunyaa كَمَآءٍkamaa'in أَنزَلْنَٰهُanzalnaahu مِنَmina ٱلسَّمَآءِs-samaa'i فَٱخْتَلَطَfakhtalata بِهِۦbihee نَبَاتُnabaatu ٱلْأَرْضِl-'ardi فَأَصْبَحَfa'asbaha هَشِيمًاhasheeman تَذْرُوهُtadhroohu ٱلرِّيَٰحُۗr-riyaahu وَكَانَwakaana ٱللَّهُl-laahu عَلَىٰ'alaa كُلِّkulli شَيْءٍshay'in مُّقْتَدِرًاmuqtadiraa٤٥ ٱلْمَالُalmaalu وَٱلْبَنُونَwalbanoona زِينَةُzeenahu ٱلْحَيَوٰةِl-hayahi ٱلدُّنْيَاۖd-dunyaa وَٱلْبَٰقِيَٰتُwalbaaqiyaatu ٱلصَّٰلِحَٰتُs-saalihaatu خَيْرٌkhayrun عِندَ'inda رَبِّكَrabbika ثَوَابًاthawaaban وَخَيْرٌwakhayrun أَمَلًاamalaa٤٦ وَيَوْمَwayawma نُسَيِّرُnusayyiru ٱلْجِبَالَl-jibaala وَتَرَىwataraa ٱلْأَرْضَl-'arda بَارِزَةًbaarizahan وَحَشَرْنَٰهُمْwahasharnaahum فَلَمْfalam نُغَادِرْnughaadir مِنْهُمْminhum أَحَدًاahadaa٤٧ وَعُرِضُواْwa'uridooaa عَلَىٰ'alaa رَبِّكَrabbika صَفًّاsaffan لَّقَدْlaqad جِئْتُمُونَاji'tumoonaa كَمَاkamaa خَلَقْنَٰكُمْkhalaqnaakum أَوَّلَawwala مَرَّةِۭۚmarrahi بَلْbal زَعَمْتُمْza'amtum أَلَّنallan نَّجْعَلَnaj'ala لَكُمlakum مَّوْعِدًاmaw'idaa٤٨ وَوُضِعَwawudi'a ٱلْكِتَٰبُl-kitaabu فَتَرَىfataraa ٱلْمُجْرِمِينَl-mujrimeena مُشْفِقِينَmushfiqeena مِمَّاmimmaa فِيهِfeehi وَيَقُولُونَwayaqooloona يَٰوَيْلَتَنَاyaawaylatanaa مَالِmaali
