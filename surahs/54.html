Tajweed keyrules in this surah
  • Madd Laazimمد لازمHold 6 counts. Obligatory.
  • Madd Muttasilمد متصلHold 4 to 5 counts.
  • Madd Munfasilمد منفصلHold 4 to 5 counts.
  • IqlabإقلابNoon turns into a meem sound.
  • Idghaam w/ Ghunnahإدغام بغنةMerge into the next letter with nasalisation.
  • Idghaam w/o Ghunnahإدغام بغير غنةMerge into the next letter, no nasalisation.
  • Idghaam Shafawiإدغام شفويMeem merges into meem.
  • Idghaam Mutajanisaynإدغام متجانسينLetters of the same articulation merge.
  • Ikhfa Shafawiإخفاء شفويHide the meem before a baa.
  • IkhfaإخفاءHide the noon with a light nasal sound.
  • GhunnahغنةNasal sound, 2 counts.
  • QalqalahقلقلةEcho the letter with a slight bounce.
  • Lam Shamsiyyahلام شمسيةSilent laam, merged into the next letter.
  • Hamzat al-Waslهمزة وصلSkipped when joined to the word before.
  • Madd Tabee'iمد طبيعيNatural madd. Hold 2 counts.
Al-Qamar القمر Meccan · 55 ayat
ٱقْتَرَبَتِiqtarabati ٱلسَّاعَةُs-saa'ahu وَٱنشَقَّwanshaqqa ٱلْقَمَرُl-qamar١ وَإِنwa'in يَرَوْاْyarawaa ءَايَةًaayahan يُعْرِضُواْyu'ridooaa وَيَقُولُواْwayaqoolooaa سِحْرٌsihrun مُّسْتَمِرٌّmustamirr٢ وَكَذَّبُواْwakadhdhabooaa وَٱتَّبَعُوٓاْwattaba'ooaa أَهْوَآءَهُمْۚahwaa'ahum وَكُلُّwakullu أَمْرٍamrin مُّسْتَقِرٌّmustaqirr٣ وَلَقَدْwalaqad جَآءَهُمjaa'ahum مِّنَmina ٱلْأَنۢبَآءِl-'anbaa'i مَاmaa فِيهِfeehi مُزْدَجَرٌmuzdajar٤ حِكْمَةُۢhikmahu بَٰلِغَةٌۖbaalighahun فَمَاfamaa تُغْنِtughni ٱلنُّذُرُn-nudhur٥ فَتَوَلَّfatawalla عَنْهُمْۘ'anhum يَوْمَyawma يَدْعُyad'u ٱلدَّاعِd-daa'i إِلَىٰilaa شَيْءٍshay'in نُّكُرٍnukur٦ خُشَّعًاkhushsha'an أَبْصَٰرُهُمْabsaaruhum يَخْرُجُونَyakhrujoona مِنَmina ٱلْأَجْدَاثِl-'ajdaathi كَأَنَّهُمْka'annahum جَرَادٌjaraadun مُّنتَشِرٌmuntashir٧ مُّهْطِعِينَmmuhti'eena إِلَىilaa ٱلدَّاعِۖd-daa'i يَقُولُyaqoolu ٱلْكَٰفِرُونَl-kaafiroona هَٰذَاhaadhaa يَوْمٌyawmun عَسِرٌ'asir٨ ۞كَذَّبَتْkadhdhabat قَبْلَهُمْqablahum قَوْمُqawmu نُوحٍnoohin فَكَذَّبُواْfakadhdhabooaa عَبْدَنَا'abdanaa وَقَالُواْwaqaalooaa مَجْنُونٌmajnoonun وَٱزْدُجِرَwazdujir٩ فَدَعَاfada'aa رَبَّهُۥٓrabbahoo أَنِّيannee مَغْلُوبٌmaghloobun فَٱنتَصِرْfantasir١٠ فَفَتَحْنَآfafatahnaa أَبْوَٰبَabwaaba ٱلسَّمَآءِs-samaa'i بِمَآءٍbimaa'in مُّنْهَمِرٍmunhamir١١ وَفَجَّرْنَاwafajjarnaa ٱلْأَرْضَl-'arda عُيُونًا'uyoonan فَٱلْتَقَىfaltaqaa ٱلْمَآءُl-maa'u عَلَىٰٓ'alaa أَمْرٍamrin قَدْqad قُدِرَqudir١٢ وَحَمَلْنَٰهُwahamalnaahu عَلَىٰ'alaa ذَاتِdhaati أَلْوَٰحٍalwaahin وَدُسُرٍwadusur١٣ تَجْرِيtajree بِأَعْيُنِنَاbi'a'yuninaa جَزَآءًjazaa'an لِّمَنliman كَانَkaana كُفِرَkufir١٤ وَلَقَدwalaqad تَّرَكْنَٰهَآtaraknaahaa ءَايَةًaayahan فَهَلْfahal مِنmin مُّدَّكِرٍmuddakir١٥ فَكَيْفَfakayfa كَانَkaana عَذَابِي'adhaabee وَنُذُرِwanudhur١٦ وَلَقَدْwalaqad يَسَّرْنَاyassarnaa ٱلْقُرْءَانَl-qur'aana لِلذِّكْرِlidhdhikri فَهَلْfahal مِنmin مُّدَّكِرٍmuddakir١٧ كَذَّبَتْkadhdhabat عَادٌ'aadun فَكَيْفَfakayfa كَانَkaana عَذَابِي'adhaabee وَنُذُرِwanudhur١٨ إِنَّآinnaa أَرْسَلْنَاarsalnaa عَلَيْهِمْ'alayhim رِيحًاreehan صَرْصَرًاsarsaran فِيfee يَوْمِyawmi نَحْسٍnahsin مُّسْتَمِرٍّmustamirr١٩ تَنزِعُtanzi'u ٱلنَّاسَn-naasa كَأَنَّهُمْka'annahum أَعْجَازُa'jaazu نَخْلٍnakhlin مُّنقَعِرٍmunqa'ir٢٠ فَكَيْفَfakayfa كَانَkaana عَذَابِي'adhaabee وَنُذُرِwanudhur٢١ وَلَقَدْwalaqad يَسَّرْنَاyassarnaa ٱلْقُرْءَانَl-qur'aana لِلذِّكْرِlidhdhikri فَهَلْfahal مِنmin مُّدَّكِرٍmuddakir٢٢ كَذَّبَتْkadhdhabat ثَمُودُthamoodu بِٱلنُّذُرِbinnudhur٢٣ فَقَالُوٓاْfaqaalooaa أَبَشَرًاabasharan مِّنَّاminnaa وَٰحِدًاwaahidan نَّتَّبِعُهُۥٓnattabi'uhoo إِنَّآinnaa إِذًاidhan لَّفِيlafee ضَلَٰلٍdalaalin وَسُعُرٍwasu'ur٢٤ أَءُلْقِيَa'ulqiya ٱلذِّكْرُdh-dhikru عَلَيْهِ'alayhi مِنۢmin بَيْنِنَاbayninaa بَلْbal هُوَhuwa كَذَّابٌkadhdhaabun أَشِرٌashir٢٥ سَيَعْلَمُونَsaya'lamoona غَدًاghadan مَّنِmani ٱلْكَذَّابُl-kadhdhaabu ٱلْأَشِرُl-'ashir٢٦ إِنَّاinnaa مُرْسِلُواْmursilooaa ٱلنَّاقَةِn-naaqahi فِتْنَةًfitnahan لَّهُمْlahum فَٱرْتَقِبْهُمْfartaqibhum وَٱصْطَبِرْwastabir٢٧ وَنَبِّئْهُمْwanabbi'hum أَنَّanna ٱلْمَآءَl-maa'a قِسْمَةُۢqismahu بَيْنَهُمْۖbaynahum كُلُّkullu شِرْبٍshirbin مُّحْتَضَرٌmuhtadar٢٨ فَنَادَوْاْfanaadawaa صَاحِبَهُمْsaahibahum فَتَعَاطَىٰfata'aataa فَعَقَرَfa'aqar٢٩ فَكَيْفَfakayfa كَانَkaana عَذَابِي'adhaabee وَنُذُرِwanudhur٣٠ إِنَّآinnaa أَرْسَلْنَاarsalnaa عَلَيْهِمْ'alayhim صَيْحَةًsayhahan وَٰحِدَةًwaahidahan فَكَانُواْfakaanooaa كَهَشِيمِkahasheemi ٱلْمُحْتَظِرِl-muhtazir٣١ وَلَقَدْwalaqad يَسَّرْنَاyassarnaa ٱلْقُرْءَانَl-qur'aana لِلذِّكْرِlidhdhikri فَهَلْfahal مِنmin مُّدَّكِرٍmuddakir٣٢ كَذَّبَتْkadhdhabat قَوْمُqawmu لُوطِۭlooti بِٱلنُّذُرِbinnudhur٣٣ إِنَّآinnaa أَرْسَلْنَاarsalnaa عَلَيْهِمْ'alayhim حَاصِبًاhaasiban إِلَّآillaa ءَالَaala لُوطٍۖlootin نَّجَّيْنَٰهُمnajjaynaahum بِسَحَرٍbisahar٣٤ نِّعْمَةًnni'mahan مِّنْmin عِندِنَاۚ'indinaa كَذَٰلِكَkadhaalika نَجْزِيnajzee مَنman شَكَرَshakar٣٥ وَلَقَدْwalaqad أَنذَرَهُمandharahum بَطْشَتَنَاbatshatanaa فَتَمَارَوْاْfatamaarawaa بِٱلنُّذُرِbinnudhur٣٦ وَلَقَدْwalaqad رَٰوَدُوهُraawadoohu عَن'an ضَيْفِهِۦdayfihee فَطَمَسْنَآfatamasnaa أَعْيُنَهُمْa'yunahum فَذُوقُواْfadhooqooaa عَذَابِي'adhaabee وَنُذُرِwanudhur٣٧ وَلَقَدْwalaqad صَبَّحَهُمsabbahahum بُكْرَةًbukrahan عَذَابٌ'adhaabun مُّسْتَقِرٌّmustaqirr٣٨ فَذُوقُواْfadhooqooaa عَذَابِي'adhaabee وَنُذُرِwanudhur٣٩ وَلَقَدْwalaqad يَسَّرْنَاyassarnaa ٱلْقُرْءَانَl-qur'aana لِلذِّكْرِlidhdhikri فَهَلْfahal مِنmin مُّدَّكِرٍmuddakir٤٠ وَلَقَدْwalaqad جَآءَjaa'a ءَالَaala فِرْعَوْنَfir'awna ٱلنُّذُرُn-nudhur٤١ كَذَّبُواْkadhdhabooaa بِئَايَٰتِنَاbi'aayaatinaa كُلِّهَاkullihaa فَأَخَذْنَٰهُمْfa'akhadhnaahum أَخْذَakhdha عَزِيزٍ'azeezin مُّقْتَدِرٍmuqtadir٤٢ أَكُفَّارُكُمْakuffaarukum خَيْرٌkhayrun مِّنْmin أُوْلَٰٓئِكُمْoolaa'ikum أَمْam لَكُمlakum بَرَآءَةٌbaraa'ahun فِيfee ٱلزُّبُرِz-zubur٤٣ أَمْam يَقُولُونَyaqooloona نَحْنُnahnu جَمِيعٌjamee'un مُّنتَصِرٌmuntasir٤٤ سَيُهْزَمُsayuhzamu ٱلْجَمْعُl-jam'u وَيُوَلُّونَwayuwalloona ٱلدُّبُرَd-dubur٤٥ بَلِbali ٱلسَّاعَةُs-saa'ahu مَوْعِدُهُمْmaw'iduhum وَٱلسَّاعَةُwassaa'ahu أَدْهَىٰadhaa وَأَمَرُّwa'amarr٤٦ إِنَّinna ٱلْمُجْرِمِينَl-mujrimeena فِيfee ضَلَٰلٍdalaalin وَسُعُرٍwasu'ur٤٧ يَوْمَyawma يُسْحَبُونَyushaboona فِيfee ٱلنَّارِn-naari عَلَىٰ'alaa وُجُوهِهِمْwujoohihim ذُوقُواْdhooqooaa مَسَّmassa سَقَرَsaqar٤٨ إِنَّاinnaa كُلَّkulla شَيْءٍshay'in خَلَقْنَٰهُkhalaqnaahu بِقَدَرٍbiqadar٤٩ وَمَآwamaa أَمْرُنَآamrunaa إِلَّاillaa وَٰحِدَةٌwaahidahun كَلَمْحِۭkalamhi بِٱلْبَصَرِbilbasar٥٠ وَلَقَدْwalaqad أَهْلَكْنَآahlaknaa أَشْيَاعَكُمْashyaa'akum فَهَلْfahal مِنmin مُّدَّكِرٍmuddakir٥١ وَكُلُّwakullu شَيْءٍshay'in فَعَلُوهُfa'aloohu فِيfee ٱلزُّبُرِz-zubur٥٢ وَكُلُّwakullu صَغِيرٍsagheerin وَكَبِيرٍwakabeerin مُّسْتَطَرٌmustatar٥٣ إِنَّinna ٱلْمُتَّقِينَl-muttaqeena فِيfee جَنَّٰتٍjannaatin وَنَهَرٍwanahar٥٤ فِيfee مَقْعَدِmaq'adi صِدْقٍsidqin عِندَ'inda مَلِيكٍmaleekin مُّقْتَدِرِۭmuqtadir٥٥
