Tajweed keyrules in this surah
  • Madd Laazimمد لازمHold 6 counts. Obligatory.
  • Madd Muttasilمد متصلHold 4 to 5 counts.
  • Madd Munfasilمد منفصلHold 4 to 5 counts.
  • IqlabإقلابNoon turns into a meem sound.
  • Idghaam w/ Ghunnahإدغام بغنةMerge into the next letter with nasalisation.
  • Idghaam w/o Ghunnahإدغام بغير غنةMerge into the next letter, no nasalisation.
  • Idghaam Shafawiإدغام شفويMeem merges into meem.
  • Ikhfa Shafawiإخفاء شفويHide the meem before a baa.
  • IkhfaإخفاءHide the noon with a light nasal sound.
  • GhunnahغنةNasal sound, 2 counts.
  • QalqalahقلقلةEcho the letter with a slight bounce.
  • Lam Shamsiyyahلام شمسيةSilent laam, merged into the next letter.
  • Hamzat al-Waslهمزة وصلSkipped when joined to the word before.
  • Silent letterحرف لا ينطقWritten but not pronounced.
  • Madd Tabee'iمد طبيعيNatural madd. Hold 2 counts.
Muhammad محمد Medinan · 38 ayat
ٱلَّذِينَalladheena كَفَرُواْkafarooaa وَصَدُّواْwasaddooaa عَن'an سَبِيلِsabeeli ٱللَّهِl-laahi أَضَلَّadalla أَعْمَٰلَهُمْa'maalahum١ وَٱلَّذِينَwalladheena ءَامَنُواْaamanooaa وَعَمِلُواْwa'amilooaa ٱلصَّٰلِحَٰتِs-saalihaati وَءَامَنُواْwa'aamanooaa بِمَاbimaa نُزِّلَnuzzila عَلَىٰ'alaa مُحَمَّدٍmuhammadin وَهُوَwahuwa ٱلْحَقُّl-haqqu مِنmin رَّبِّهِمْrabbihim كَفَّرَkaffara عَنْهُمْ'anhum سَيِّئَاتِهِمْsayyi'aatihim وَأَصْلَحَwa'aslaha بَالَهُمْbaalahum٢ ذَٰلِكَdhaalika بِأَنَّbi'anna ٱلَّذِينَl-ladheena كَفَرُواْkafarooaa ٱتَّبَعُواْt-taba'ooaa ٱلْبَٰطِلَl-baatila وَأَنَّwa'anna ٱلَّذِينَl-ladheena ءَامَنُواْaamanooaa ٱتَّبَعُواْt-taba'ooaa ٱلْحَقَّl-haqqa مِنmin رَّبِّهِمْۚrabbihim كَذَٰلِكَkadhaalika يَضْرِبُyadribu ٱللَّهُl-laahu لِلنَّاسِlinnaasi أَمْثَٰلَهُمْamthaalahum٣ فَإِذَاfa'idhaa لَقِيتُمُlaqeetumu ٱلَّذِينَl-ladheena كَفَرُواْkafarooaa فَضَرْبَfadarba ٱلرِّقَابِr-riqaabi حَتَّىٰٓhattaa إِذَآidhaa أَثْخَنتُمُوهُمْathkhantumoohum فَشُدُّواْfashuddooaa ٱلْوَثَاقَl-wathaaqa فَإِمَّاfa'immaa مَنَّۢاmannaa بَعْدُba'du وَإِمَّاwa'immaa فِدَآءًfidaa'an حَتَّىٰhattaa تَضَعَtada'a ٱلْحَرْبُl-harbu أَوْزَارَهَاۚawzaarahaa ذَٰلِكَۖdhaalika وَلَوْwalaw يَشَآءُyashaa'u ٱللَّهُl-laahu لَٱنتَصَرَlantasara مِنْهُمْminhum وَلَٰكِنwalaakin لِّيَبْلُوَاْliyabluwaa بَعْضَكُمba'dakum بِبَعْضٍۗbiba'din وَٱلَّذِينَwalladheena قُتِلُواْqutilooaa فِيfee سَبِيلِsabeeli ٱللَّهِl-laahi فَلَنfalan يُضِلَّyudilla أَعْمَٰلَهُمْa'maalahum٤ سَيَهْدِيهِمْsayahdeehim وَيُصْلِحُwayuslihu بَالَهُمْbaalahum٥ وَيُدْخِلُهُمُwayudkhiluhumu ٱلْجَنَّةَl-jannaha عَرَّفَهَا'arrafahaa لَهُمْlahum٦ يَٰٓأَيُّهَاyaa'ayyuhaa ٱلَّذِينَl-ladheena ءَامَنُوٓاْaamanooaa إِنin تَنصُرُواْtansurooaa ٱللَّهَl-laaha يَنصُرْكُمْyansurkum وَيُثَبِّتْwayuthabbit أَقْدَامَكُمْaqdaamakum٧ وَٱلَّذِينَwalladheena كَفَرُواْkafarooaa فَتَعْسًاfata'san لَّهُمْlahum وَأَضَلَّwa'adalla أَعْمَٰلَهُمْa'maalahum٨ ذَٰلِكَdhaalika بِأَنَّهُمْbi'annahum كَرِهُواْkarihooaa مَآmaa أَنزَلَanzala ٱللَّهُl-laahu فَأَحْبَطَfa'ahbata أَعْمَٰلَهُمْa'maalahum٩ ۞أَفَلَمْ'afalam يَسِيرُواْyaseerooaa فِيfee ٱلْأَرْضِl-'ardi فَيَنظُرُواْfayanzurooaa كَيْفَkayfa كَانَkaana عَٰقِبَةُ'aaqibahu ٱلَّذِينَl-ladheena مِنmin قَبْلِهِمْۖqablihim دَمَّرَdammara ٱللَّهُl-laahu عَلَيْهِمْۖ'alayhim وَلِلْكَٰفِرِينَwalilkaafireena أَمْثَٰلُهَاamthaaluhaa١٠ ذَٰلِكَdhaalika بِأَنَّbi'anna ٱللَّهَl-laaha مَوْلَىmawlaa ٱلَّذِينَl-ladheena ءَامَنُواْaamanooaa وَأَنَّwa'anna ٱلْكَٰفِرِينَl-kaafireena لَاlaa مَوْلَىٰmawlaa لَهُمْlahum١١ إِنَّinna ٱللَّهَl-laaha يُدْخِلُyudkhilu ٱلَّذِينَl-ladheena ءَامَنُواْaamanooaa وَعَمِلُواْwa'amilooaa ٱلصَّٰلِحَٰتِs-saalihaati جَنَّٰتٍjannaatin تَجْرِيtajree مِنmin تَحْتِهَاtahtihaa ٱلْأَنْهَٰرُۖl-'anhaaru وَٱلَّذِينَwalladheena كَفَرُواْkafarooaa يَتَمَتَّعُونَyatamatta'oona وَيَأْكُلُونَwaya'kuloona كَمَاkamaa تَأْكُلُta'kulu ٱلْأَنْعَٰمُl-'an'aamu وَٱلنَّارُwannaaru مَثْوًىmathwan لَّهُمْlahum١٢ وَكَأَيِّنwaka'ayyin مِّنmin قَرْيَةٍqaryahin هِيَhiya أَشَدُّashaddu قُوَّةًquwwahan مِّنmin قَرْيَتِكَqaryatika ٱلَّتِيٓl-latee أَخْرَجَتْكَakhrajatka أَهْلَكْنَٰهُمْahlaknaahum فَلَاfalaa نَاصِرَnaasira لَهُمْlahum١٣ أَفَمَنafaman كَانَkaana عَلَىٰ'alaa بَيِّنَةٍbayyinahin مِّنmin رَّبِّهِۦrabbihee كَمَنkaman زُيِّنَzuyyina لَهُۥlahoo سُوٓءُsoo'u عَمَلِهِۦ'amalihee وَٱتَّبَعُوٓاْwattaba'ooaa أَهْوَآءَهُمahwaa'ahum١٤ مَّثَلُmmathalu ٱلْجَنَّةِl-jannahi ٱلَّتِيl-latee وُعِدَwu'ida ٱلْمُتَّقُونَۖl-muttaqoona فِيهَآfeehaa أَنْهَٰرٌanhaarun مِّنmin مَّآءٍmaa'in غَيْرِghayri ءَاسِنٍaasinin وَأَنْهَٰرٌwa'anhaarun مِّنmin لَّبَنٍlabanin لَّمْlam يَتَغَيَّرْyataghayyar طَعْمُهُۥta'muhoo وَأَنْهَٰرٌwa'anhaarun مِّنْmin خَمْرٍkhamrin لَّذَّةٍladhdhahin لِّلشَّٰرِبِينَlishshaaribeena وَأَنْهَٰرٌwa'anhaarun مِّنْmin عَسَلٍ'asalin مُّصَفًّىۖmusaffan وَلَهُمْwalahum فِيهَاfeehaa مِنmin كُلِّkulli ٱلثَّمَرَٰتِth-thamaraati وَمَغْفِرَةٌwamaghfirahun مِّنmin رَّبِّهِمْۖrabbihim كَمَنْkaman هُوَhuwa خَٰلِدٌkhaalidun فِيfee ٱلنَّارِn-naari وَسُقُواْwasuqooaa مَآءًmaa'an حَمِيمًاhameeman فَقَطَّعَfaqatta'a أَمْعَآءَهُمْam'aa'ahum١٥ وَمِنْهُمwaminhum مَّنman يَسْتَمِعُyastami'u إِلَيْكَilayka حَتَّىٰٓhattaa إِذَاidhaa خَرَجُواْkharajooaa مِنْmin عِندِكَ'indika قَالُواْqaalooaa لِلَّذِينَlilladheena أُوتُواْootooaa ٱلْعِلْمَl-'ilma مَاذَاmaadhaa قَالَqaala ءَانِفًاۚaanifan أُوْلَٰٓئِكَoolaa'ika ٱلَّذِينَl-ladheena طَبَعَtaba'a ٱللَّهُl-laahu عَلَىٰ'alaa قُلُوبِهِمْquloobihim وَٱتَّبَعُوٓاْwattaba'ooaa أَهْوَآءَهُمْahwaa'ahum١٦ وَٱلَّذِينَwalladheena ٱهْتَدَوْاْh-tadawaa زَادَهُمْzaadahum هُدًىhudan وَءَاتَىٰهُمْwa'aataahum تَقْوَىٰهُمْtaqwaahum١٧ فَهَلْfahal يَنظُرُونَyanzuroona إِلَّاillaa ٱلسَّاعَةَs-saa'aha أَنan تَأْتِيَهُمta'tiyahum بَغْتَةًۖbaghtahan فَقَدْfaqad جَآءَjaa'a أَشْرَاطُهَاۚashraatuhaa فَأَنَّىٰfa'annaa لَهُمْlahum إِذَاidhaa جَآءَتْهُمْjaa'athum ذِكْرَىٰهُمْdhikraahum١٨ فَٱعْلَمْfa'lam أَنَّهُۥannahoo لَآlaa إِلَٰهَilaaha إِلَّاillaa ٱللَّهُl-laahu وَٱسْتَغْفِرْwastaghfir لِذَنۢبِكَlidhanbika وَلِلْمُؤْمِنِينَwalilmu'mineena وَٱلْمُؤْمِنَٰتِۗwalmu'minaati وَٱللَّهُwallaahu يَعْلَمُya'lamu مُتَقَلَّبَكُمْmutaqallabakum وَمَثْوَىٰكُمْwamathwaakum١٩ وَيَقُولُwayaqoolu ٱلَّذِينَl-ladheena ءَامَنُواْaamanooaa لَوْلَاlawlaa نُزِّلَتْnuzzilat سُورَةٌۖsoorahun فَإِذَآfa'idhaa أُنزِلَتْunzilat سُورَةٌsoorahun مُّحْكَمَةٌmuhkamahun وَذُكِرَwadhukira فِيهَاfeehaa ٱلْقِتَالُl-qitaalu رَأَيْتَra'ayta ٱلَّذِينَl-ladheena فِيfee قُلُوبِهِمquloobihim مَّرَضٌmaradun يَنظُرُونَyanzuroona إِلَيْكَilayka نَظَرَnazara ٱلْمَغْشِيِّl-maghshiyyi عَلَيْهِ'alayhi مِنَmina ٱلْمَوْتِۖl-mawti فَأَوْلَىٰfa'awlaa لَهُمْlahum٢٠ طَاعَةٌtaa'ahun وَقَوْلٌwaqawlun مَّعْرُوفٌۚma'roofun فَإِذَاfa'idhaa عَزَمَ'azama ٱلْأَمْرُl-'amru فَلَوْfalaw صَدَقُواْsadaqooaa ٱللَّهَl-laaha لَكَانَlakaana خَيْرًاkhayran لَّهُمْlahum٢١ فَهَلْfahal عَسَيْتُمْ'asaytum إِنin تَوَلَّيْتُمْtawallaytum أَنan تُفْسِدُواْtufsidooaa فِيfee ٱلْأَرْضِl-'ardi وَتُقَطِّعُوٓاْwatuqatti'ooaa أَرْحَامَكُمْarhaamakum٢٢ أُوْلَٰٓئِكَoolaa'ika ٱلَّذِينَl-ladheena لَعَنَهُمُla'anahumu ٱللَّهُl-laahu فَأَصَمَّهُمْfa'asammahum وَأَعْمَىٰٓwa'a'maa أَبْصَٰرَهُمْabsaarahum٢٣ أَفَلَاafalaa يَتَدَبَّرُونَyatadabbaroona ٱلْقُرْءَانَl-qur'aana أَمْam عَلَىٰ'alaa قُلُوبٍquloobin أَقْفَالُهَآaqfaaluhaa٢٤ إِنَّinna ٱلَّذِينَl-ladheena ٱرْتَدُّواْr-taddooaa عَلَىٰٓ'alaa أَدْبَٰرِهِمadbaarihim مِّنۢmin بَعْدِba'di مَاmaa تَبَيَّنَtabayyana لَهُمُlahumu ٱلْهُدَىl-hudaa ٱلشَّيْطَٰنُsh-shaytaanu سَوَّلَsawwala لَهُمْlahum وَأَمْلَىٰwa'amlaa لَهُمْlahum٢٥ ذَٰلِكَdhaalika بِأَنَّهُمْbi'annahum قَالُواْqaalooaa لِلَّذِينَlilladheena كَرِهُواْkarihooaa مَاmaa نَزَّلَnazzala ٱللَّهُl-laahu سَنُطِيعُكُمْsanutee'ukum فِيfee بَعْضِba'di ٱلْأَمْرِۖl-'amri وَٱللَّهُwallaahu يَعْلَمُya'lamu إِسْرَارَهُمْisraarahum٢٦ فَكَيْفَfakayfa إِذَاidhaa تَوَفَّتْهُمُtawaffathumu ٱلْمَلَٰٓئِكَةُl-malaa'ikahu يَضْرِبُونَyadriboona وُجُوهَهُمْwujoohahum وَأَدْبَٰرَهُمْwa'adbaarahum٢٧ ذَٰلِكَdhaalika بِأَنَّهُمُbi'annahumu ٱتَّبَعُواْt-taba'ooaa مَآmaa أَسْخَطَaskhata ٱللَّهَl-laaha وَكَرِهُواْwakarihooaa رِضْوَٰنَهُۥridwaanahoo فَأَحْبَطَfa'ahbata أَعْمَٰلَهُمْa'maalahum٢٨ أَمْam حَسِبَhasiba ٱلَّذِينَl-ladheena فِيfee قُلُوبِهِمquloobihim مَّرَضٌmaradun أَنan لَّنlan يُخْرِجَyukhrija ٱللَّهُl-laahu أَضْغَٰنَهُمْadghaanahum٢٩ وَلَوْwalaw نَشَآءُnashaa'u لَأَرَيْنَٰكَهُمْla'araynaakahum فَلَعَرَفْتَهُمfala'araftahum بِسِيمَٰهُمْۚbiseemaahum وَلَتَعْرِفَنَّهُمْwalata'rifannahum فِيfee لَحْنِlahni ٱلْقَوْلِۚl-qawli وَٱللَّهُwallaahu يَعْلَمُya'lamu أَعْمَٰلَكُمْa'maalakum٣٠ وَصَدُّواْwasaddooaa عَن'an سَبِيلِsabeeli ٱللَّهِl-laahi ثُمَّthumma مَاتُواْmaatooaa وَهُمْwahum كُفَّارٌkuffaarun فَلَنfalan يَغْفِرَyaghfira ٱللَّهُl-laahu لَهُمْlahum٣٤ فَلَاfalaa تَهِنُواْtahinooaa وَتَدْعُوٓاْwatad'ooaa إِلَىilaa ٱلسَّلْمِs-salmi وَأَنتُمُwa'antumu ٱلْأَعْلَوْنَl-'a'lawna وَٱللَّهُwallaahu مَعَكُمْma'akum وَلَنwalan يَتِرَكُمْyatirakum أَعْمَٰلَكُمْa'maalakum٣٥ إِنَّمَاinnamaa ٱلْحَيَوٰةُl-hayahu ٱلدُّنْيَاd-dunyaa لَعِبٌla'ibun وَلَهْوٌۚwalahwun وَإِنwa'in تُؤْمِنُواْtu'minooaa وَتَتَّقُواْwatattaqooaa يُؤْتِكُمْyu'tikum أُجُورَكُمْujoorakum وَلَاwalaa يَسْئَلْكُمْyas'alkum أَمْوَٰلَكُمْamwaalakum٣٦ إِنin يَسْئَلْكُمُوهَاyas'alkumoohaa فَيُحْفِكُمْfayuhfikum تَبْخَلُواْtabkhalooaa وَيُخْرِجْwayukhrij أَضْغَٰنَكُمْadghaanakum٣٧ هَٰٓأَنتُمْhaa'antum هَٰٓؤُلَآءِhaa'ulaa'i تُدْعَوْنَtud'awna لِتُنفِقُواْlitunfiqooaa فِيfee سَبِيلِsabeeli ٱللَّهِl-laahi فَمِنكُمfaminkum مَّنman يَبْخَلُۖyabkhalu وَمَنwaman يَبْخَلْyabkhal فَإِنَّمَاfa'innamaa يَبْخَلُyabkhalu عَن'an نَّفْسِهِۦۚnafsihee وَٱللَّهُwallaahu ٱلْغَنِيُّl-ghaniyyu وَأَنتُمُwa'antumu ٱلْفُقَرَآءُۚl-fuqaraa'u وَإِنwa'in تَتَوَلَّوْاْtatawallawaa يَسْتَبْدِلْyastabdil قَوْمًاqawman غَيْرَكُمْghayrakum ثُمَّthumma لَاlaa يَكُونُوٓاْyakoonooaa أَمْثَٰلَكُمamthaalakum٣٨
