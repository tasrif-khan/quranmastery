Tajweed keyrules in this surah
  • Madd Laazimمد لازمHold 6 counts. Obligatory.
  • Madd Muttasilمد متصلHold 4 to 5 counts.
  • Madd Munfasilمد منفصلHold 4 to 5 counts.
  • Madd al-Aaridمد عارضHold 2, 4 or 6 counts when stopping.
  • IqlabإقلابNoon turns into a meem sound.
  • Idghaam w/ Ghunnahإدغام بغنةMerge into the next letter with nasalisation.
  • Idghaam w/o Ghunnahإدغام بغير غنةMerge into the next letter, no nasalisation.
  • Idghaam Shafawiإدغام شفويMeem merges into meem.
  • Ikhfa Shafawiإخفاء شفويHide the meem before a baa.
  • IkhfaإخفاءHide the noon with a light nasal sound.
  • GhunnahغنةNasal sound, 2 counts.
  • QalqalahقلقلةEcho the letter with a slight bounce.
  • Lam Shamsiyyahلام شمسيةSilent laam, merged into the next letter.
  • Hamzat al-Waslهمزة وصلSkipped when joined to the word before.
  • Silent letterحرف لا ينطقWritten but not pronounced.
  • Madd Tabee'iمد طبيعيNatural madd. Hold 2 counts.

نَصِيرٍnaseer٣١ وَمِنْwamin ءَايَٰتِهِaayaatihi ٱلْجَوَارِl-jawaari فِيfee ٱلْبَحْرِl-bahri كَٱلْأَعْلَٰمِkal'a'laam٣٢ إِنin يَشَأْyasha' يُسْكِنِyuskini ٱلرِّيحَr-reeha فَيَظْلَلْنَfayazlalna رَوَاكِدَrawaakida عَلَىٰ'alaa ظَهْرِهِۦٓۚzahrihee إِنَّinna فِيfee ذَٰلِكَdhaalika لَأٓيَٰتٍla'yaatin لِّكُلِّlikulli صَبَّارٍsabbaarin شَكُورٍshakoor٣٣ أَوْaw يُوبِقْهُنَّyoobiqhunna بِمَاbimaa كَسَبُواْkasabooaa وَيَعْفُwaya'fu عَن'an كَثِيرٍkatheer٣٤ وَيَعْلَمَwaya'lama ٱلَّذِينَl-ladheena يُجَٰدِلُونَyujaadiloona فِيٓfee ءَايَٰتِنَاaayaatinaa مَاmaa لَهُمlahum مِّنmin مَّحِيصٍmahees٣٥ فَمَآfamaa أُوتِيتُمooteetum مِّنmin شَيْءٍshay'in فَمَتَٰعُfamataa'u ٱلْحَيَوٰةِl-hayahi ٱلدُّنْيَاۚd-dunyaa وَمَاwamaa عِندَ'inda ٱللَّهِl-laahi خَيْرٌkhayrun وَأَبْقَىٰwa'abqaa لِلَّذِينَlilladheena ءَامَنُواْaamanooaa وَعَلَىٰwa'alaa رَبِّهِمْrabbihim يَتَوَكَّلُونَyatawakkaloon٣٦ وَٱلَّذِينَwalladheena يَجْتَنِبُونَyajtaniboona كَبَٰٓئِرَkabaa'ira ٱلْإِثْمِl-'ithmi وَٱلْفَوَٰحِشَwalfawaahisha وَإِذَاwa'idhaa مَاmaa غَضِبُواْghadibooaa هُمْhum يَغْفِرُونَyaghfiroon٣٧ وَٱلَّذِينَwalladheena ٱسْتَجَابُواْs-tajaabooaa لِرَبِّهِمْlirabbihim وَأَقَامُواْwa'aqaamooaa ٱلصَّلَوٰةَs-salaha وَأَمْرُهُمْwa'amruhum شُورَىٰshooraa بَيْنَهُمْbaynahum وَمِمَّاwamimmaa رَزَقْنَٰهُمْrazaqnaahum يُنفِقُونَyunfiqoon٣٨ وَٱلَّذِينَwalladheena إِذَآidhaa أَصَابَهُمُasaabahumu ٱلْبَغْيُl-baghyu هُمْhum يَنتَصِرُونَyantasiroon٣٩ وَجَزَٰٓؤُاْwajazaa'uaa سَيِّئَةٍsayyi'ahin سَيِّئَةٌsayyi'ahun مِّثْلُهَاۖmithluhaa فَمَنْfaman عَفَا'afaa وَأَصْلَحَwa'aslaha فَأَجْرُهُۥfa'ajruhoo عَلَى'alaa ٱللَّهِۚl-laahi إِنَّهُۥinnahoo لَاlaa يُحِبُّyuhibbu ٱلظَّٰلِمِينَz-zaalimeen٤٠ وَلَمَنِwalamani ٱنتَصَرَn-tasara بَعْدَba'da ظُلْمِهِۦzulmihee فَأُوْلَٰٓئِكَfa'oolaa'ika مَاmaa عَلَيْهِم'alayhim مِّنmin سَبِيلٍsabeel٤١ إِنَّمَاinnamaa ٱلسَّبِيلُs-sabeelu عَلَى'alaa ٱلَّذِينَl-ladheena يَظْلِمُونَyazlimoona ٱلنَّاسَn-naasa وَيَبْغُونَwayabghoona فِيfee ٱلْأَرْضِl-'ardi بِغَيْرِbighayri ٱلْحَقِّۚl-haqqi أُوْلَٰٓئِكَoolaa'ika لَهُمْlahum عَذَابٌ'adhaabun أَلِيمٌaleem٤٢ وَلَمَنwalaman صَبَرَsabara وَغَفَرَwaghafara إِنَّinna ذَٰلِكَdhaalika لَمِنْlamin عَزْمِ'azmi ٱلْأُمُورِl-'umoor٤٣ وَمَنwaman يُضْلِلِyudlili ٱللَّهُl-laahu فَمَاfamaa لَهُۥlahoo مِنmin وَلِيٍّwaliyyin مِّنۢmin بَعْدِهِۦۗba'dihee وَتَرَىwataraa ٱلظَّٰلِمِينَz-zaalimeena لَمَّاlammaa رَأَوُاْra'awuaa ٱلْعَذَابَl-'adhaaba يَقُولُونَyaqooloona هَلْhal إِلَىٰilaa مَرَدٍّmaraddin مِّنmin سَبِيلٍsabeel٤٤ وَتَرَىٰهُمْwataraahum يُعْرَضُونَyu'radoona عَلَيْهَا'alayhaa خَٰشِعِينَkhaashi'eena مِنَmina ٱلذُّلِّdh-dhulli يَنظُرُونَyanzuroona مِنmin طَرْفٍtarfin خَفِيٍّۗkhafiyyin وَقَالَwaqaala ٱلَّذِينَl-ladheena ءَامَنُوٓاْaamanooaa إِنَّinna ٱلْخَٰسِرِينَl-khaasireena ٱلَّذِينَl-ladheena خَسِرُوٓاْkhasirooaa أَنفُسَهُمْanfusahum وَأَهْلِيهِمْwa'ahleehim يَوْمَyawma ٱلْقِيَٰمَةِۗl-qiyaamahi أَلَآalaa إِنَّinna ٱلظَّٰلِمِينَz-zaalimeena فِيfee عَذَابٍ'adhaabin مُّقِيمٍmuqeem٤٥ وَمَاwamaa كَانَkaana لَهُمlahum مِّنْmin أَوْلِيَآءَawliyaa'a يَنصُرُونَهُمyansuroonahum مِّنmin دُونِdooni ٱللَّهِۗl-laahi وَمَنwaman يُضْلِلِyudlili ٱللَّهُl-laahu فَمَاfamaa لَهُۥlahoo مِنmin سَبِيلٍsabeel٤٦ ٱسْتَجِيبُواْistajeebooaa لِرَبِّكُمlirabbikum مِّنmin قَبْلِqabli أَنan يَأْتِيَya'tiya يَوْمٌyawmun لَّاlaa مَرَدَّmaradda لَهُۥlahoo مِنَmina ٱللَّهِۚl-laahi مَاmaa لَكُمlakum مِّنmin مَّلْجَإٍmalja'in يَوْمَئِذٍyawma'idhin وَمَاwamaa لَكُمlakum مِّنmin نَّكِيرٍnakeer٤٧ فَإِنْfa'in أَعْرَضُواْa'radooaa فَمَآfamaa أَرْسَلْنَٰكَarsalnaaka عَلَيْهِمْ'alayhim حَفِيظًاۖhafeezan إِنْin عَلَيْكَ'alayka إِلَّاillaa ٱلْبَلَٰغُۗl-balaaghu وَإِنَّآwa'innaa إِذَآidhaa أَذَقْنَاadhaqnaa ٱلْإِنسَٰنَl-'insaana مِنَّاminnaa رَحْمَةًrahmahan فَرِحَfariha بِهَاۖbihaa وَإِنwa'in تُصِبْهُمْtusibhum سَيِّئَةُۢsayyi'ahu بِمَاbimaa قَدَّمَتْqaddamat أَيْدِيهِمْaydeehim فَإِنَّfa'inna ٱلْإِنسَٰنَl-'insaana كَفُورٌkafoor٤٨ لِّلَّهِllillaahi مُلْكُmulku ٱلسَّمَٰوَٰتِs-samaawaati وَٱلْأَرْضِۚwal'ardi يَخْلُقُyakhluqu مَاmaa يَشَآءُۚyashaa'u يَهَبُyahabu لِمَنliman يَشَآءُyashaa'u إِنَٰثًاinaathan وَيَهَبُwayahabu لِمَنliman يَشَآءُyashaa'u ٱلذُّكُورَdh-dhukoor٤٩ أَوْaw يُزَوِّجُهُمْyuzawwijuhum ذُكْرَانًاdhukraanan وَإِنَٰثًاۖwa'inaathan وَيَجْعَلُwayaj'alu مَنman يَشَآءُyashaa'u عَقِيمًاۚ'aqeeman إِنَّهُۥinnahoo عَلِيمٌ'aleemun قَدِيرٌqadeer٥٠ ۞وَمَاwamaa كَانَkaana لِبَشَرٍlibasharin أَنan يُكَلِّمَهُyukallimahu ٱللَّهُl-laahu إِلَّاillaa وَحْيًاwahyan أَوْaw مِنmin وَرَآيِٕwaraayi حِجَابٍhijaabin أَوْaw يُرْسِلَyursila رَسُولًاrasoolan فَيُوحِيَfayoohiya بِإِذْنِهِۦbi'idhnihee مَاmaa يَشَآءُۚyashaa'u إِنَّهُۥinnahoo عَلِيٌّ'aliyyun حَكِيمٌhakeem٥١ وَكَذَٰلِكَwakadhaalika أَوْحَيْنَآawhaynaa إِلَيْكَilayka رُوحًاroohan مِّنْmin أَمْرِنَاۚamrinaa مَاmaa كُنتَkunta تَدْرِيtadree مَاmaa ٱلْكِتَٰبُl-kitaabu وَلَاwalaa ٱلْإِيمَٰنُl-'eemaanu وَلَٰكِنwalaakin جَعَلْنَٰهُja'alnaahu نُورًاnooran نَّهْدِيnahdee بِهِۦbihee مَنman نَّشَآءُnashaa'u مِنْmin عِبَادِنَاۚ'ibaadinaa وَإِنَّكَwa'innaka لَتَهْدِيٓlatahdee إِلَىٰilaa صِرَٰطٍsiraatin مُّسْتَقِيمٍmustaqeem٥٢ صِرَٰطِsiraati ٱللَّهِl-laahi ٱلَّذِيl-ladhee لَهُۥlahoo مَاmaa فِيfee ٱلسَّمَٰوَٰتِs-samaawaati وَمَاwamaa فِيfee ٱلْأَرْضِۗl-'ardi أَلَآalaa إِلَىilaa ٱللَّهِl-laahi تَصِيرُtaseeru ٱلْأُمُورُl-'umoor٥٣
