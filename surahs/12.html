وَكَأَيِّنwaka'ayyin مِّنْmin ءَايَةٍaayahin فِيfee ٱلسَّمَٰوَٰتِs-samaawaati وَٱلْأَرْضِwal'ardi يَمُرُّونَyamurroona عَلَيْهَا'alayhaa وَهُمْwahum عَنْهَا'anhaa مُعْرِضُونَmu'ridoon١٠٥ وَمَاwamaa يُؤْمِنُyu'minu أَكْثَرُهُمaktharuhum بِٱللَّهِbillaahi إِلَّاillaa وَهُمwahum مُّشْرِكُونَmushrikoon١٠٦ أَفَأَمِنُوٓاْafa'aminooaa أَنan تَأْتِيَهُمْta'tiyahum غَٰشِيَةٌghaashiyahun مِّنْmin عَذَابِ'adhaabi ٱللَّهِl-laahi أَوْaw تَأْتِيَهُمُta'tiyahumu ٱلسَّاعَةُs-saa'ahu بَغْتَةًbaghtahan وَهُمْwahum لَاlaa يَشْعُرُونَyash'uroon١٠٧ قُلْqul هَٰذِهِۦhaadhihee سَبِيلِيٓsabeelee أَدْعُوٓاْad'ooaa إِلَىilaa ٱللَّهِۚl-laahi عَلَىٰ'alaa بَصِيرَةٍbaseerahin أَنَا۠anaa وَمَنِwamani ٱتَّبَعَنِيۖt-taba'anee وَسُبْحَٰنَwasubhaana ٱللَّهِl-laahi وَمَآwamaa أَنَا۠anaa مِنَmina ٱلْمُشْرِكِينَl-mushrikeen١٠٨ وَمَآwamaa أَرْسَلْنَاarsalnaa مِنmin قَبْلِكَqablika إِلَّاillaa رِجَالًاrijaalan نُّوحِيٓnoohee إِلَيْهِمilayhim مِّنْmin أَهْلِahli ٱلْقُرَىٰٓۗl-quraa أَفَلَمْafalam يَسِيرُواْyaseerooaa فِيfee ٱلْأَرْضِl-'ardi فَيَنظُرُواْfayanzurooaa كَيْفَkayfa كَانَkaana عَٰقِبَةُ'aaqibahu ٱلَّذِينَl-ladheena مِنmin قَبْلِهِمْۗqablihim وَلَدَارُwaladaaru ٱلْأٓخِرَةِl-'khirahi خَيْرٌkhayrun لِّلَّذِينَlilladheena ٱتَّقَوْاْۚt-taqawaa أَفَلَاafalaa تَعْقِلُونَta'qiloon١٠٩ حَتَّىٰٓhattaa إِذَاidhaa ٱسْتَيْئَسَs-tay'asa ٱلرُّسُلُr-rusulu وَظَنُّوٓاْwazannooaa أَنَّهُمْannahum قَدْqad كُذِبُواْkudhibooaa جَآءَهُمْjaa'ahum نَصْرُنَاnasrunaa فَنُجِّيَfanujjiya مَنman نَّشَآءُۖnashaa'u وَلَاwalaa يُرَدُّyuraddu بَأْسُنَاba'sunaa عَنِ'ani ٱلْقَوْمِl-qawmi ٱلْمُجْرِمِينَl-mujrimeen١١٠ لَقَدْlaqad كَانَkaana فِيfee قَصَصِهِمْqasasihim عِبْرَةٌ'ibrahun لِّأُوْلِيli'oolee ٱلْأَلْبَٰبِۗl-'albaabi مَاmaa كَانَkaana حَدِيثًاhadeethan يُفْتَرَىٰyuftaraa وَلَٰكِنwalaakin تَصْدِيقَtasdeeqa ٱلَّذِيl-ladhee بَيْنَbayna يَدَيْهِyadayhi وَتَفْصِيلَwatafseela كُلِّkulli شَيْءٍshay'in وَهُدًىwahudan وَرَحْمَةًwarahmahan لِّقَوْمٍliqawmin
