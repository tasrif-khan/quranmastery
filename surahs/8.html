Tajweed keyrules in this surah
  • Madd Laazimمد لازمHold 6 counts. Obligatory.
  • Madd Muttasilمد متصلHold 4 to 5 counts.
  • Madd Munfasilمد منفصلHold 4 to 5 counts.
  • Madd al-Aaridمد عارضHold 2, 4 or 6 counts when stopping.
  • IqlabإقلابNoon turns into a meem sound.
  • Idghaam w/ Ghunnahإدغام بغنةMerge into the next letter with nasalisation.
  • Idghaam w/o Ghunnahإدغام بغير غنةMerge into the next letter, no nasalisation.
  • Idghaam Shafawiإدغام شفويMeem merges into meem.
  • Idghaam Mutajanisaynإدغام متجانسينLetters of the same articulation merge.
  • Ikhfa Shafawiإخفاء شفويHide the meem before a baa.
  • IkhfaإخفاءHide the noon with a light nasal sound.
  • GhunnahغنةNasal sound, 2 counts.
  • QalqalahقلقلةEcho the letter with a slight bounce.
  • Lam Shamsiyyahلام شمسيةSilent laam, merged into the next letter.
  • Hamzat al-Waslهمزة وصلSkipped when joined to the word before.
  • Silent letterحرف لا ينطقWritten but not pronounced.
  • Madd Tabee'iمد طبيعيNatural madd. Hold 2 counts.

ٱللَّهِۚl-laahi فَسَيُنفِقُونَهَاfasayunfiqoonahaa ثُمَّthumma تَكُونُtakoonu عَلَيْهِمْ'alayhim حَسْرَةًhasrahan ثُمَّthumma يُغْلَبُونَۗyughlaboona وَٱلَّذِينَwalladheena كَفَرُوٓاْkafarooaa إِلَىٰilaa جَهَنَّمَjahannama يُحْشَرُونَyuhsharoon٣٦ لِيَمِيزَliyameeza ٱللَّهُl-laahu ٱلْخَبِيثَl-khabeetha مِنَmina ٱلطَّيِّبِt-tayyibi وَيَجْعَلَwayaj'ala ٱلْخَبِيثَl-khabeetha بَعْضَهُۥba'dahoo عَلَىٰ'alaa بَعْضٍba'din فَيَرْكُمَهُۥfayarkumahoo جَمِيعًاjamee'an فَيَجْعَلَهُۥfayaj'alahoo فِيfee جَهَنَّمَۚjahannama أُوْلَٰٓئِكَoolaa'ika هُمُhumu ٱلْخَٰسِرُونَl-khaasiroon٣٧ قُلqul لِّلَّذِينَlilladheena كَفَرُوٓاْkafarooaa إِنin يَنتَهُواْyantahooaa يُغْفَرْyughfar لَهُمlahum مَّاmaa قَدْqad سَلَفَsalafa وَإِنwa'in يَعُودُواْya'oodooaa فَقَدْfaqad مَضَتْmadat سُنَّتُsunnatu ٱلْأَوَّلِينَl-'awwaleen٣٨ وَقَٰتِلُوهُمْwaqaatiloohum حَتَّىٰhattaa لَاlaa تَكُونَtakoona فِتْنَةٌfitnahun وَيَكُونَwayakoona ٱلدِّينُd-deenu كُلُّهُۥkulluhoo لِلَّهِۚlillaahi فَإِنِfa'ini ٱنتَهَوْاْn-tahawaa فَإِنَّfa'inna ٱللَّهَl-laaha بِمَاbimaa يَعْمَلُونَya'maloona بَصِيرٌbaseer٣٩ وَإِنwa'in تَوَلَّوْاْtawallawaa فَٱعْلَمُوٓاْfa'lamooaa أَنَّanna ٱللَّهَl-laaha مَوْلَىٰكُمْۚmawlaakum نِعْمَni'ma ٱلْمَوْلَىٰl-mawlaa وَنِعْمَwani'ma ٱلنَّصِيرُn-naseer٤٠ ۞وَٱعْلَمُوٓاْwa'lamooaa أَنَّمَاannamaa غَنِمْتُمghanimtum مِّنmin شَيْءٍshay'in فَأَنَّfa'anna لِلَّهِlillaahi خُمُسَهُۥkhumusahoo وَلِلرَّسُولِwalirrasooli وَلِذِيwalidhee ٱلْقُرْبَىٰl-qurbaa وَٱلْيَتَٰمَىٰwalyataamaa وَٱلْمَسَٰكِينِwalmasaakeeni وَٱبْنِwabni ٱلسَّبِيلِs-sabeeli إِنin كُنتُمْkuntum ءَامَنتُمaamantum بِٱللَّهِbillaahi وَمَآwamaa أَنزَلْنَاanzalnaa عَلَىٰ'alaa عَبْدِنَا'abdinaa يَوْمَyawma ٱلْفُرْقَانِl-furqaani يَوْمَyawma ٱلْتَقَىl-taqaa ٱلْجَمْعَانِۗl-jam'aani وَٱللَّهُwallaahu عَلَىٰ'alaa كُلِّkulli شَيْءٍshay'in قَدِيرٌqadeer٤١ إِذْidh أَنتُمantum بِٱلْعُدْوَةِbil'udwahi ٱلدُّنْيَاd-dunyaa وَهُمwahum بِٱلْعُدْوَةِbil'udwahi ٱلْقُصْوَىٰl-quswaa وَٱلرَّكْبُwarrakbu أَسْفَلَasfala مِنكُمْۚminkum وَلَوْwalaw تَوَاعَدتُّمْtawaa'adttum لَٱخْتَلَفْتُمْlakhtalaftum فِيfee ٱلْمِيعَٰدِl-mee'aadi وَلَٰكِنwalaakin لِّيَقْضِيَliyaqdiya ٱللَّهُl-laahu أَمْرًاamran كَانَkaana مَفْعُولًاmaf'oolan لِّيَهْلِكَliyahlika مَنْman هَلَكَhalaka عَنۢ'an بَيِّنَةٍbayyinahin وَيَحْيَىٰwayahyaa مَنْman حَيَّhayya عَنۢ'an بَيِّنَةٍۗbayyinahin وَإِنَّwa'inna ٱللَّهَl-laaha لَسَمِيعٌlasamee'un عَلِيمٌ'aleem٤٢ إِذْidh يُرِيكَهُمُyureekahumu ٱللَّهُl-laahu فِيfee مَنَامِكَmanaamika قَلِيلًاۖqaleelan وَلَوْwalaw أَرَىٰكَهُمْaraakahum كَثِيرًاkatheeran لَّفَشِلْتُمْlafashiltum وَلَتَنَٰزَعْتُمْwalatanaaza'tum فِيfee ٱلْأَمْرِl-'amri وَلَٰكِنَّwalaakinna ٱللَّهَl-laaha سَلَّمَۚsallama إِنَّهُۥinnahoo عَلِيمُۢ'aleemu بِذَاتِbidhaati ٱلصُّدُورِs-sudoor٤٣ وَإِذْwa'idh يُرِيكُمُوهُمْyureekumoohum إِذِidhi ٱلْتَقَيْتُمْl-taqaytum فِيٓfee أَعْيُنِكُمْa'yunikum قَلِيلًاqaleelan وَيُقَلِّلُكُمْwayuqallilukum فِيٓfee أَعْيُنِهِمْa'yunihim لِيَقْضِيَliyaqdiya ٱللَّهُl-laahu أَمْرًاamran كَانَkaana مَفْعُولًاۗmaf'oolan وَإِلَىwa'ilaa ٱللَّهِl-laahi تُرْجَعُturja'u ٱلْأُمُورُl-'umoor٤٤ يَٰٓأَيُّهَاyaa'ayyuhaa ٱلَّذِينَl-ladheena ءَامَنُوٓاْaamanooaa إِذَاidhaa لَقِيتُمْlaqeetum فِئَةًfi'ahan فَٱثْبُتُواْfathbutooaa وَٱذْكُرُواْwadhkurooaa ٱللَّهَl-laaha كَثِيرًاkatheeran لَّعَلَّكُمْla'allakum تُفْلِحُونَtuflihoon٤٥ وَأَطِيعُواْwa'atee'ooaa ٱللَّهَl-laaha وَرَسُولَهُۥwarasoolahoo وَلَاwalaa تَنَٰزَعُواْtanaaza'ooaa فَتَفْشَلُواْfatafshalooaa وَتَذْهَبَwatadhhaba رِيحُكُمْۖreehukum وَٱصْبِرُوٓاْۚwasbirooaa إِنَّinna ٱللَّهَl-laaha مَعَma'a ٱلصَّٰبِرِينَs-saabireen٤٦ وَلَاwalaa تَكُونُواْtakoonooaa كَٱلَّذِينَkalladheena خَرَجُواْkharajooaa مِنmin دِيَٰرِهِمdiyaarihim بَطَرًاbataran وَرِئَآءَwari'aa'a ٱلنَّاسِn-naasi وَيَصُدُّونَwayasuddoona عَن'an سَبِيلِsabeeli ٱللَّهِۚl-laahi وَٱللَّهُwallaahu بِمَاbimaa يَعْمَلُونَya'maloona مُحِيطٌmuheet٤٧ وَإِذْwa'idh زَيَّنَzayyana لَهُمُlahumu ٱلشَّيْطَٰنُsh-shaytaanu أَعْمَٰلَهُمْa'maalahum وَقَالَwaqaala لَاlaa غَالِبَghaaliba لَكُمُlakumu ٱلْيَوْمَl-yawma مِنَmina ٱلنَّاسِn-naasi وَإِنِّيwa'innee جَارٌjaarun لَّكُمْۖlakum فَلَمَّاfalammaa تَرَآءَتِtaraa'ati ٱلْفِئَتَانِl-fi'ataani نَكَصَnakasa عَلَىٰ'alaa عَقِبَيْهِ'aqibayhi وَقَالَwaqaala إِنِّيinnee بَرِيٓءٌbaree'un مِّنكُمْminkum إِنِّيٓinnee أَرَىٰaraa مَاmaa لَاlaa تَرَوْنَtarawna إِنِّيٓinnee أَخَافُakhaafu ٱللَّهَۚl-laaha وَٱللَّهُwallaahu شَدِيدُshadeedu ٱلْعِقَابِl-'iqaab٤٨ إِذْidh يَقُولُyaqoolu ٱلْمُنَٰفِقُونَl-munaafiqoona وَٱلَّذِينَwalladheena فِيfee قُلُوبِهِمquloobihim مَّرَضٌmaradun غَرَّgharra هَٰٓؤُلَآءِhaa'ulaa'i دِينُهُمْۗdeenuhum وَمَنwaman يَتَوَكَّلْyatawakkal عَلَى'alaa ٱللَّهِl-laahi فَإِنَّfa'inna ٱللَّهَl-laaha عَزِيزٌ'azeezun حَكِيمٌhakeem٤٩ وَلَوْwalaw تَرَىٰٓtaraa إِذْidh يَتَوَفَّىyatawaffaa ٱلَّذِينَl-ladheena كَفَرُواْkafarooaa ٱلْمَلَٰٓئِكَةُl-malaa'ikahu يَضْرِبُونَyadriboona بِأَنَّbi'anna ٱللَّهَl-laaha لَمْlam يَكُyaku مُغَيِّرًاmughayyiran نِّعْمَةًni'mahan أَنْعَمَهَاan'amahaa عَلَىٰ'alaa قَوْمٍqawmin حَتَّىٰhattaa يُغَيِّرُواْyughayyirooaa مَاmaa إِنَّهُۥinnahoo عَزِيزٌ'azeezun حَكِيمٌhakeem٦٣ يَٰٓأَيُّهَاyaa'ayyuhaa ٱلنَّبِيُّn-nabiyyu حَسْبُكَhasbuka ٱللَّهُl-laahu وَمَنِwamani ٱتَّبَعَكَt-taba'aka مِنَmina ٱلْمُؤْمِنِينَl-mu'mineen٦٤ يَٰٓأَيُّهَاyaa'ayyuhaa ٱلنَّبِيُّn-nabiyyu حَرِّضِharridi ٱلْمُؤْمِنِينَl-mu'mineena عَلَى'alaa ٱلْقِتَالِۚl-qitaali إِنin يَكُنyakun مِّنكُمْminkum عِشْرُونَ'ishroona صَٰبِرُونَsaabiroona يَغْلِبُواْyaghlibooaa مِاْئَتَيْنِۚmiaa'atayni وَإِنwa'in يَكُنyakun مِّنكُمminkum مِّاْئَةٌmiaa'ahun يَغْلِبُوٓاْyaghlibooaa أَلْفًاalfan مِّنَmina ٱلَّذِينَl-ladheena كَفَرُواْkafarooaa بِأَنَّهُمْbi'annahum قَوْمٌqawmun لَّاlaa يَفْقَهُونَyafqahoon٦٥ ٱلْئَٰنَal'aana خَفَّفَkhaffafa ٱللَّهُl-laahu عَنكُمْ'ankum وَعَلِمَwa'alima أَنَّanna فِيكُمْfeekum ضَعْفًاۚda'fan فَإِنfa'in يَكُنyakun مِّنكُمminkum مِّاْئَةٌmiaa'ahun صَابِرَةٌsaabirahun يَغْلِبُواْyaghlibooaa مِاْئَتَيْنِۚmiaa'atayni وَإِنwa'in يَكُنyakun مِّنكُمْminkum أَلْفٌalfun يَغْلِبُوٓاْyaghlibooaa أَلْفَيْنِalfayni بِإِذْنِbi'idhni ٱللَّهِۗl-laahi وَٱللَّهُwallaahu مَعَma'a ٱلصَّٰبِرِينَs-saabireen٦٦ مَاmaa كَانَkaana لِنَبِيٍّlinabiyyin أَنan يَكُونَyakoona لَهُۥٓlahoo أَسْرَىٰasraa حَتَّىٰhattaa يُثْخِنَyuthkhina فِيfee ٱلْأَرْضِۚl-'ardi تُرِيدُونَtureedoona عَرَضَ'arada ٱلدُّنْيَاd-dunyaa وَٱللَّهُwallaahu يُرِيدُyureedu ٱلْأٓخِرَةَۗl-'khiraha وَٱللَّهُwallaahu عَزِيزٌ'azeezun حَكِيمٌhakeem٦٧ لَّوْلَاllawlaa كِتَٰبٌkitaabun مِّنَmina ٱللَّهِl-laahi سَبَقَsabaqa لَمَسَّكُمْlamassakum فِيمَآfeemaa أَخَذْتُمْakhadhtum عَذَابٌ'adhaabun عَظِيمٌ'azeem٦٨ فَكُلُواْfakulooaa مِمَّاmimmaa غَنِمْتُمْghanimtum حَلَٰلًاhalaalan طَيِّبًاۚtayyiban وَٱتَّقُواْwattaqooaa ٱللَّهَۚl-laaha إِنَّinna ٱللَّهَl-laaha غَفُورٌghafoorun رَّحِيمٌraheem٦٩ يَٰٓأَيُّهَاyaa'ayyuhaa ٱلنَّبِيُّn-nabiyyu قُلqul لِّمَنliman فِيٓfee أَيْدِيكُمaydeekum مِّنَmina ٱلْأَسْرَىٰٓl-'asraa إِنin يَعْلَمِya'lami ٱللَّهُl-laahu فِيfee قُلُوبِكُمْquloobikum خَيْرًاkhayran يُؤْتِكُمْyu'tikum خَيْرًاkhayran مِّمَّآmimmaa أُخِذَukhidha مِنكُمْminkum وَيَغْفِرْwayaghfir لَكُمْۚlakum وَٱللَّهُwallaahu غَفُورٌghafoorun رَّحِيمٌraheem٧٠ وَإِنwa'in يُرِيدُواْyureedooaa خِيَانَتَكَkhiyaanataka فَقَدْfaqad خَانُواْkhaanooaa ٱللَّهَl-laaha مِنmin قَبْلُqablu فَأَمْكَنَfa'amkana مِنْهُمْۗminhum وَٱللَّهُwallaahu عَلِيمٌ'aleemun حَكِيمٌhakeem٧١ إِنَّinna ٱلَّذِينَl-ladheena ءَامَنُواْaamanooaa وَهَاجَرُواْwahaajarooaa وَجَٰهَدُواْwajaahadooaa بِأَمْوَٰلِهِمْbi'amwaalihim وَأَنفُسِهِمْwa'anfusihim فِيfee سَبِيلِsabeeli ٱللَّهِl-laahi وَٱلَّذِينَwalladheena ءَاوَواْaawawaa وَّنَصَرُوٓاْwanasarooaa أُوْلَٰٓئِكَoolaa'ika بَعْضُهُمْba'duhum أَوْلِيَآءُawliyaa'u بَعْضٍۚba'din وَٱلَّذِينَwalladheena ءَامَنُواْaamanooaa وَلَمْwalam يُهَاجِرُواْyuhaajirooaa مَاmaa لَكُمlakum مِّنmin وَلَٰيَتِهِمwalaayatihim مِّنmin شَيْءٍshay'in حَتَّىٰhattaa يُهَاجِرُواْۚyuhaajirooaa وَإِنِwa'ini ٱسْتَنصَرُوكُمْs-tansarookum فِيfee ٱلدِّينِd-deeni فَعَلَيْكُمُfa'alaykumu ٱلنَّصْرُn-nasru إِلَّاillaa عَلَىٰ'alaa قَوْمِۭqawmi بَيْنَكُمْbaynakum وَبَيْنَهُمwabaynahum مِّيثَٰقٌۗmeethaaqun وَٱللَّهُwallaahu بِمَاbimaa تَعْمَلُونَta'maloona بَصِيرٌbaseer٧٢ وَٱلَّذِينَwalladheena كَفَرُواْkafarooaa بَعْضُهُمْba'duhum أَوْلِيَآءُawliyaa'u بَعْضٍۚba'din إِلَّاillaa تَفْعَلُوهُtaf'aloohu تَكُنtakun فِتْنَةٌfitnahun فِيfee ٱلْأَرْضِl-'ardi وَفَسَادٌwafasaadun كَبِيرٌkabeer٧٣ وَٱلَّذِينَwalladheena ءَامَنُواْaamanooaa وَهَاجَرُواْwahaajarooaa وَجَٰهَدُواْwajaahadooaa فِيfee سَبِيلِsabeeli ٱللَّهِl-laahi وَٱلَّذِينَwalladheena ءَاوَواْaawawaa وَّنَصَرُوٓاْwanasarooaa أُوْلَٰٓئِكَoolaa'ika هُمُhumu ٱلْمُؤْمِنُونَl-mu'minoona حَقًّاۚhaqqan لَّهُمlahum مَّغْفِرَةٌmaghfirahun وَرِزْقٌwarizqun كَرِيمٌkareem٧٤ وَٱلَّذِينَwalladheena ءَامَنُواْaamanooaa مِنۢmin بَعْدُba'du وَهَاجَرُواْwahaajarooaa وَجَٰهَدُواْwajaahadooaa مَعَكُمْma'akum فَأُوْلَٰٓئِكَfa'oolaa'ika مِنكُمْۚminkum وَأُوْلُواْwa'oolooaa ٱلْأَرْحَامِl-'arhaami بَعْضُهُمْba'duhum أَوْلَىٰawlaa بِبَعْضٍbiba'din فِيfee كِتَٰبِkitaabi ٱللَّهِۚl-llaahi إِنَّinna ٱللَّهَl-llaaha بِكُلِّbikulli شَيْءٍshay'in عَلِيمُۢ'aleem٧٥
