Tajweed keyrules in this surah
  • Madd Laazimمد لازمHold 6 counts. Obligatory.
  • Madd Muttasilمد متصلHold 4 to 5 counts.
  • Madd Munfasilمد منفصلHold 4 to 5 counts.
  • Madd al-Aaridمد عارضHold 2, 4 or 6 counts when stopping.
  • IqlabإقلابNoon turns into a meem sound.
  • Idghaam w/ Ghunnahإدغام بغنةMerge into the next letter with nasalisation.
  • Idghaam w/o Ghunnahإدغام بغير غنةMerge into the next letter, no nasalisation.
  • Idghaam Shafawiإدغام شفويMeem merges into meem.
  • Ikhfa Shafawiإخفاء شفويHide the meem before a baa.
  • IkhfaإخفاءHide the noon with a light nasal sound.
  • GhunnahغنةNasal sound, 2 counts.
  • QalqalahقلقلةEcho the letter with a slight bounce.
  • Lam Shamsiyyahلام شمسيةSilent laam, merged into the next letter.
  • Hamzat al-Waslهمزة وصلSkipped when joined to the word before.
  • Silent letterحرف لا ينطقWritten but not pronounced.
  • Madd Tabee'iمد طبيعيNatural madd. Hold 2 counts.

ٱلدُّنْيَاd-dunyaa وَمَاwamaa ٱلْحَيَوٰةُl-hayahu ٱلدُّنْيَاd-dunyaa فِيfee ٱلْأٓخِرَةِl-'khirahi إِلَّاillaa مَتَٰعٌmataa'٢٦ وَيَقُولُwayaqoolu ٱلَّذِينَl-ladheena كَفَرُواْkafarooaa لَوْلَآlawlaa أُنزِلَunzila عَلَيْهِ'alayhi ءَايَةٌaayahun مِّنmin رَّبِّهِۦۚrabbihee قُلْqul إِنَّinna ٱللَّهَl-laaha يُضِلُّyudillu مَنman يَشَآءُyashaa'u وَيَهْدِيٓwayahdee إِلَيْهِilayhi مَنْman أَنَابَanaab٢٧ ٱلَّذِينَalladheena ءَامَنُواْaamanooaa وَتَطْمَئِنُّwatatma'innu قُلُوبُهُمquloobuhum بِذِكْرِbidhikri ٱللَّهِۗl-laahi أَلَاalaa بِذِكْرِbidhikri ٱللَّهِl-laahi تَطْمَئِنُّtatma'innu ٱلْقُلُوبُl-quloob٢٨ ٱلَّذِينَalladheena ءَامَنُواْaamanooaa وَعَمِلُواْwa'amilooaa ٱلصَّٰلِحَٰتِs-saalihaati طُوبَىٰtoobaa لَهُمْlahum وَحُسْنُwahusnu مَئَابٍma'aab٢٩ كَذَٰلِكَkadhaalika أَرْسَلْنَٰكَarsalnaaka فِيٓfee أُمَّةٍummahin قَدْqad خَلَتْkhalat مِنmin قَبْلِهَآqablihaa أُمَمٌumamun لِّتَتْلُوَاْlitatluwaa عَلَيْهِمُ'alayhimu ٱلَّذِيٓl-ladhee أَوْحَيْنَآawhaynaa إِلَيْكَilayka وَهُمْwahum يَكْفُرُونَyakfuroona بِٱلرَّحْمَٰنِۚbirrahmaani قُلْqul هُوَhuwa رَبِّيrabbee لَآlaa إِلَٰهَilaaha إِلَّاillaa هُوَhuwa عَلَيْهِ'alayhi تَوَكَّلْتُtawakkaltu وَإِلَيْهِwa'ilayhi مَتَابِmataab٣٠ وَلَوْwalaw أَنَّanna قُرْءَانًاqur'aanan سُيِّرَتْsuyyirat بِهِbihi ٱلْجِبَالُl-jibaalu أَوْaw قُطِّعَتْqutti'at بِهِbihi ٱلْأَرْضُl-'ardu أَوْaw كُلِّمَkullima بِهِbihi ٱلْمَوْتَىٰۗl-mawtaa بَلbal لِّلَّهِlillaahi ٱلْأَمْرُl-'amru جَمِيعًاۗjamee'an أَفَلَمْafalam يَاْيْئَسِyaay'asi ٱلَّذِينَl-ladheena ءَامَنُوٓاْaamanooaa أَنan لَّوْlaw يَشَآءُyashaa'u ٱللَّهُl-laahu لَهَدَىlahadaa ٱلنَّاسَn-naasa جَمِيعًاۗjamee'an وَلَاwalaa يَزَالُyazaalu ٱلَّذِينَl-ladheena كَفَرُواْkafarooaa تُصِيبُهُمtuseebuhum بِمَاbimaa صَنَعُواْsana'ooaa قَارِعَةٌqaari'ahun أَوْaw تَحُلُّtahullu قَرِيبًاqareeban مِّنmin دَارِهِمْdaarihim حَتَّىٰhattaa يَأْتِيَya'tiya وَعْدُwa'du ٱللَّهِۚl-laahi إِنَّinna ٱللَّهَl-laaha لَاlaa يُخْلِفُyukhlifu ٱلْمِيعَادَl-mee'aad٣١ وَلَقَدِwalaqadi ٱسْتُهْزِئَs-tuhzi'a بِرُسُلٍbirusulin مِّنmin قَبْلِكَqablika فَأَمْلَيْتُfa'amlaytu لِلَّذِينَlilladheena كَفَرُواْkafarooaa ثُمَّthumma أَخَذْتُهُمْۖakhadhtuhum فَكَيْفَfakayfa كَانَkaana عِقَابِ'iqaab٣٢ أَفَمَنْafaman هُوَhuwa قَآئِمٌqaa'imun عَلَىٰ'alaa كُلِّkulli نَفْسِۭnafsi بِمَاbimaa كَسَبَتْۗkasabat وَجَعَلُواْwaja'alooaa لِلَّهِlillaahi شُرَكَآءَshurakaa'a قُلْqul سَمُّوهُمْۚsammoohum أَمْam تُنَبِّئُونَهُۥtunabbi'oonahoo بِمَاbimaa لَاlaa يَعْلَمُya'lamu فِيfee ٱلْأَرْضِl-'ardi أَمam بِظَٰهِرٍbizaahirin مِّنَmina ٱلْقَوْلِۗl-qawli بَلْbal زُيِّنَzuyyina لِلَّذِينَlilladheena كَفَرُواْkafarooaa مَكْرُهُمْmakruhum وَصُدُّواْwasuddooaa عَنِ'ani ٱلسَّبِيلِۗs-sabeeli وَمَنwaman يُضْلِلِyudlili ٱللَّهُl-laahu فَمَاfamaa لَهُۥlahoo مِنْmin هَادٍhaad٣٣ لَّهُمْllahum عَذَابٌ'adhaabun فِيfee ٱلْحَيَوٰةِl-hayahi ٱلدُّنْيَاۖd-dunyaa وَلَعَذَابُwala'adhaabu ٱلْأٓخِرَةِl-'khirahi أَشَقُّۖashaqqu وَمَاwamaa لَهُمlahum مِّنَmina ٱللَّهِl-laahi مِنmin وَاقٍwaaq٣٤ ۞مَّثَلُmmathalu ٱلْجَنَّةِl-jannahi ٱلَّتِيl-latee وُعِدَwu'ida ٱلْمُتَّقُونَۖl-muttaqoona تَجْرِيtajree مِنmin تَحْتِهَاtahtihaa ٱلْأَنْهَٰرُۖl-'anhaaru أُكُلُهَاukuluhaa دَآئِمٌdaa'imun وَظِلُّهَاۚwazilluhaa تِلْكَtilka عُقْبَى'uqbaa ٱلَّذِينَl-ladheena ٱتَّقَواْۚt-taqawaa وَّعُقْبَىwa'uqbaa ٱلْكَٰفِرِينَl-kaafireena ٱلنَّارُn-naar٣٥ وَٱلَّذِينَwalladheena ءَاتَيْنَٰهُمُaataynaahumu ٱلْكِتَٰبَl-kitaaba يَفْرَحُونَyafrahoona بِمَآbimaa أُنزِلَunzila إِلَيْكَۖilayka وَمِنَwamina ٱلْأَحْزَابِl-'ahzaabi مَنman يُنكِرُyunkiru بَعْضَهُۥۚba'dahoo قُلْqul إِنَّمَآinnamaa أُمِرْتُumirtu أَنْan أَعْبُدَa'buda ٱللَّهَl-laaha وَلَآwalaa أُشْرِكَushrika بِهِۦٓۚbihee إِلَيْهِilayhi أَدْعُواْad'ooaa وَإِلَيْهِwa'ilayhi مَئَابِma'aab٣٦ وَكَذَٰلِكَwakadhaalika أَنزَلْنَٰهُanzalnaahu حُكْمًاhukman عَرَبِيًّاۚ'arabiyyan وَلَئِنِwala'ini ٱتَّبَعْتَt-taba'ta أَهْوَآءَهُمahwaa'ahum بَعْدَba'da مَاmaa جَآءَكَjaa'aka مِنَmina ٱلْعِلْمِl-'ilmi مَاmaa لَكَlaka مِنَmina ٱللَّهِl-laahi مِنmin وَلِيٍّwaliyyin وَلَاwalaa وَاقٍwaaq٣٧ وَلَقَدْwalaqad أَرْسَلْنَاarsalnaa رُسُلًاrusulan مِّنmin قَبْلِكَqablika وَجَعَلْنَاwaja'alnaa لَهُمْlahum أَزْوَٰجًاazwaajan وَذُرِّيَّةًۚwadhurriyyahan وَمَاwamaa كَانَkaana لِرَسُولٍlirasoolin أَنan يَأْتِيَya'tiya بِئَايَةٍbi'aayahin إِلَّاillaa بِإِذْنِbi'idhni ٱللَّهِۗl-laahi لِكُلِّlikulli أَجَلٍajalin كِتَابٌkitaab٣٨ يَمْحُواْyamhooaa ٱللَّهُl-laahu مَاmaa يَشَآءُyashaa'u وَيُثْبِتُۖwayuthbitu وَعِندَهُۥٓwa'indahoo أُمُّummu ٱلْكِتَٰبِl-kitaab٣٩ وَإِنwa'in مَّاmaa نُرِيَنَّكَnuriyannaka بَعْضَba'da ٱلَّذِيl-ladhee نَعِدُهُمْna'iduhum أَوْaw نَتَوَفَّيَنَّكَnatawaffayannaka فَإِنَّمَاfa'innamaa عَلَيْكَ'alayka ٱلْبَلَٰغُl-balaaghu وَعَلَيْنَاwa'alaynaa ٱلْحِسَابُl-hisaab٤٠ أَوَلَمْawalam يَرَوْاْyarawaa أَنَّاannaa نَأْتِيna'tee ٱلْأَرْضَl-'arda نَنقُصُهَاnanqusuhaa مِنْmin أَطْرَافِهَاۚatraafihaa وَٱللَّهُwallaahu يَحْكُمُyahkumu لَاlaa مُعَقِّبَmu'aqqiba لِحُكْمِهِۦۚlihukmihee وَهُوَwahuwa سَرِيعُsaree'u ٱلْحِسَابِl-hisaab٤١ وَقَدْwaqad مَكَرَmakara ٱلَّذِينَl-ladheena مِنmin قَبْلِهِمْqablihim فَلِلَّهِfalillaahi ٱلْمَكْرُl-makru جَمِيعًاۖjamee'an يَعْلَمُya'lamu مَاmaa تَكْسِبُtaksibu كُلُّkullu نَفْسٍۗnafsin وَسَيَعْلَمُwasaya'lamu ٱلْكُفَّٰرُl-kuffaaru لِمَنْliman عُقْبَى'uqbaa ٱلدَّارِd-daar٤٢ وَيَقُولُwayaqoolu ٱلَّذِينَl-ladheena كَفَرُواْkafarooaa لَسْتَlasta مُرْسَلًاۚmursalan قُلْqul كَفَىٰkafaa بِٱللَّهِbillaahi شَهِيدَۢاshaheedaa بَيْنِيbaynee وَبَيْنَكُمْwabaynakum وَمَنْwaman عِندَهُۥ'indahoo عِلْمُ'ilmu ٱلْكِتَٰبِl-kitaab٤٣
